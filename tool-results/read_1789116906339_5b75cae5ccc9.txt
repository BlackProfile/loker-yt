     1→"use client";
     2→
     3→import { useEffect, useRef, useState } from "react";
     4→import type { ChangeEvent, DragEvent, FormEvent } from "react";
     5→import { toast } from "sonner";
     6→import {
     7→  CheckCircle2,
     8→  Copy,
     9→  FileText,
    10→  Loader2,
    11→  Mic,

    13→  Upload,

    17→import { Button } from "@/components/ui/button";
    18→import { Card } from "@/components/ui/card";
    19→import { Input } from "@/components/ui/input";
    20→import { Label } from "@/components/ui/label";
    21→import { Textarea } from "@/components/ui/textarea";
    22→import {
    23→  Select,

    28→} from "@/components/ui/select";

    32→
    33→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    34→const DRAFT_KEY = "lumina-draft";
    35→const AUTOSAVE_DELAY_MS = 500;
    36→const MIN_TEXT_LENGTH = 10;
    37→
    38→type FormValues = {
    39→  name: string;
    40→  email: string;
    41→  phone: string;
    42→  portfolioUrl: string;
    43→  socialLinks: string;
    44→  experience: string;
    45→  motivation: string;
    46→};
    47→
    48→type FieldKey = keyof FormValues | "positionId";
    49→type FormErrors = Partial<Record<FieldKey, string>>;
    50→
    51→const INITIAL_VALUES: FormValues = {
    52→  name: "",
    53→  email: "",
    54→  phone: "",
    55→  portfolioUrl: "",
    56→  socialLinks: "",
    57→  experience: "",
    58→  motivation: "",
    59→};
    60→
    61→const VALUE_KEYS: (keyof FormValues)[] = [
    62→  "name",
    63→  "email",
    64→  "phone",
    65→  "portfolioUrl",
    66→  "socialLinks",
    67→  "experience",
    68→  "motivation",
    69→];
    70→
    71→type StoredDraft = {
    72→  values?: Partial<Record<keyof FormValues, unknown>>;
    73→  positionId?: unknown;
    74→  savedAt?: unknown;
    75→};
    76→
    77→type ApplyWizardProps = {
    78→  positions: Position[];
    79→  positionId: string;
    80→  onPositionIdChange: (positionId: string) => void;
    81→};
    82→
    83→export function ApplyWizard({
    84→  positions,
    85→  positionId,
    86→  onPositionIdChange,
    87→}: ApplyWizardProps) {
    88→  const { t } = useLang();
    89→  const [values, setValues] = useState<FormValues>(INITIAL_VALUES);
    90→  const [errors, setErrors] = useState<FormErrors>({});
    91→  const [step, setStep] = useState(0);
    92→  const [submitting, setSubmitting] = useState(false);
    93→  const [success, setSuccess] = useState<{ name: string; trackingCode: string } | null>(
    94→    null,
    95→  );
    96→
    97→  // File unggahan (tidak masuk draft).
    98→  const [cvFile, setCvFile] = useState<File | null>(null);
    99→  const [introFile, setIntroFile] = useState<File | null>(null);
   100→  const [cvError, setCvError] = useState<string | null>(null);
   101→  const [introError, setIntroError] = useState<string | null>(null);
   102→  const [cvDragging, setCvDragging] = useState(false);
   103→  const [introDragging, setIntroDragging] = useState(false);
   104→
   105→  // Draft autosave.
   106→  const [draft, setDraft] = useState<StoredDraft | null>(null);
   107→  const submittedRef = useRef(false);
   108→  const draftDismissedRef = useRef(false);
   109→
   110→  useEffect(() => {
   111→    try {
   112→      const raw = window.localStorage.getItem(DRAFT_KEY);
   113→      if (!raw) return;
   114→      const parsed = JSON.parse(raw) as StoredDraft;
   115→      if (parsed && typeof parsed === "object" && parsed.values) setDraft(parsed);
   116→    } catch {
   117→      // draft rusak -> abaikan
   118→    }
   119→  }, []);
   120→
   121→  useEffect(() => {
   122→    if (submittedRef.current || draftDismissedRef.current) return;
   123→    const hasContent =
   124→      VALUE_KEYS.some((key) => values[key].trim().length > 0) || positionId;
   125→    if (!hasContent) return;
   126→    const timer = window.setTimeout(() => {
   127→      try {
   128→        window.localStorage.setItem(
   129→          DRAFT_KEY,
   130→          JSON.stringify({ values, positionId, savedAt: Date.now() }),
   131→        );
   132→      } catch {
   133→        // penyimpanan penuh / tidak tersedia
   134→      }
   135→    }, AUTOSAVE_DELAY_MS);
   136→    return () => window.clearTimeout(timer);
   137→  }, [values, positionId]);
   138→
   139→  const setField = (key: keyof FormValues, value: string) => {
   140→    setValues((prev) => ({ ...prev, [key]: value }));
   141→    setErrors((prev) => (prev[key] ? { ...prev, [key]: undefined } : prev));
   142→  };
   143→
   144→  const clearPositionError = () => {
   145→    setErrors((prev) =>
   146→      prev.positionId ? { ...prev, positionId: undefined } : prev,
   147→    );
   148→  };
   149→
   150→  function validateStep1(): FormErrors {
   151→    const next: FormErrors = {};
   152→    if (!values.name.trim()) next.name = t.apply.errors.name;
   153→    if (!values.email.trim()) next.email = t.apply.errors.emailRequired;
   154→    else if (!EMAIL_RE.test(values.email.trim())) next.email = t.apply.errors.emailInvalid;
   155→    if (!values.phone.trim()) next.phone = t.apply.errors.phoneRequired;
   156→    else if (values.phone.replace(/\D/g, "").length < 8)
   157→      next.phone = t.apply.errors.phoneMin;
   158→    if (positions.length > 0 && !positionId) next.positionId = t.apply.errors.position;
   159→    return next;
   160→  }
   161→
   162→  function validateStep2(): FormErrors {
   163→    const next: FormErrors = {};
   164→    if (values.experience.trim().length < MIN_TEXT_LENGTH)
   165→      next.experience = t.apply.errors.experience;
   166→    if (values.motivation.trim().length < MIN_TEXT_LENGTH)
   167→      next.motivation = t.apply.errors.motivation;
   168→    return next;
   169→  }
   170→
   171→  function goNext() {
   172→    const next = step === 0 ? validateStep1() : validateStep2();
   173→    setErrors((prev) => ({ ...prev, ...next }));
   174→    if (Object.values(next).some(Boolean)) return;
   175→    setStep((s) => Math.min(2, s + 1));
   176→  }
   177→
   178→  function goBack() {
   179→    setStep((s) => Math.max(0, s - 1));
   180→  }
   181→
   182→  function acceptCv(file: File | null) {
   183→    if (!file) return;
   184→    if (file.type !== "application/pdf") {
   185→      setCvError(t.apply.errors.cvType);
   186→      toast.error(t.apply.errors.cvType);
   187→      return;
   188→    }
   189→    if (file.size > CV_MAX_BYTES) {
   190→      setCvError(t.apply.errors.cvSize);
   191→      toast.error(t.apply.errors.cvSize);
   192→      return;
   193→    }
   194→    setCvFile(file);
   195→    setCvError(null);
   196→  }
   197→
   198→  function acceptIntro(file: File | null) {
   199→    if (!file) return;
   200→    const typeOk =
   201→      file.type.startsWith("audio/") || /\.(mp3|wav|m4a)$/i.test(file.name);
   202→    if (!typeOk) {
   203→      setIntroError(t.apply.errors.introType);
   204→      toast.error(t.apply.errors.introType);
   205→      return;
   206→    }
   207→    if (file.size > INTRO_MAX_BYTES) {
   208→      setIntroError(t.apply.errors.introSize);
   209→      toast.error(t.apply.errors.introSize);
   210→      return;
   211→    }
   212→    setIntroFile(file);
   213→    setIntroError(null);
   214→  }
   215→
   216→  function onCvInput(event: ChangeEvent<HTMLInputElement>) {
   217→    acceptCv(event.target.files?.[0] ?? null);
   218→    event.target.value = "";
   219→  }
   220→
   221→  function onIntroInput(event: ChangeEvent<HTMLInputElement>) {
   222→    acceptIntro(event.target.files?.[0] ?? null);
   223→    event.target.value = "";
   224→  }
   225→
   226→  function onDropFactory(
   227→    acceptor: (file: File | null) => void,
   228→    setDragging: (value: boolean) => void,
   229→  ) {
   230→    return (event: DragEvent<HTMLLabelElement>) => {
   231→      event.preventDefault();
   232→      setDragging(false);
   233→      acceptor(event.dataTransfer.files?.[0] ?? null);
   234→    };
   235→  }
   236→
   237→  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
   238→    event.preventDefault();
   239→    const e1 = validateStep1();
   240→    const e2 = validateStep2();
   241→    const all = { ...e1, ...e2 };
   242→    setErrors(all);
   243→    if (e1.name || e1.email || e1.phone || e1.positionId) {
   244→      setStep(0);
   245→      return;
   246→    }
   247→    if (Object.values(all).some(Boolean)) {
   248→      setStep(1);
   249→      return;
   250→    }
   251→
   252→    setSubmitting(true);
   253→    try {
   254→      const fd = new FormData();
   255→      fd.append("name", values.name.trim());
   256→      fd.append("email", values.email.trim());
   257→      fd.append("phone", values.phone.trim());
   258→      if (positionId) fd.append("positionId", positionId);
   259→      if (values.portfolioUrl.trim())
   260→        fd.append("portfolioUrl", values.portfolioUrl.trim());
   261→      if (values.socialLinks.trim())
   262→        fd.append("socialLinks", values.socialLinks.trim());
   263→      fd.append("experience", values.experience.trim());
   264→      fd.append("motivation", values.motivation.trim());
   265→      if (cvFile) fd.append("cvFile", cvFile);
   266→      if (introFile) fd.append("introFile", introFile);
   267→
   268→      const res = await fetch("/api/applications", { method: "POST", body: fd });
   269→      const data: unknown = await res.json().catch(() => null);
   270→      if (!res.ok) {
   271→        const serverError =
   272→          typeof data === "object" &&
   273→          data !== null &&
   274→          "error" in data &&
   275→          typeof (data as { error: unknown }).error === "string"
   276→            ? (data as { error: string }).error
   277→            : null;
   278→        toast.error(serverError ?? t.apply.errors.submitFailed);
   279→        return;
   280→      }
   281→      const trackingCode =
   282→        typeof data === "object" &&
   283→        data !== null &&
   284→        "trackingCode" in data &&
   285→        typeof (data as { trackingCode: unknown }).trackingCode === "string"
   286→          ? (data as { trackingCode: string }).trackingCode
   287→          : "";
   288→
   289→      submittedRef.current = true;
   290→      try {
   291→        window.localStorage.removeItem(DRAFT_KEY);
   292→      } catch {
   293→        // abaikan
   294→      }
   295→      setSuccess({ name: values.name.trim(), trackingCode });
   296→      toast.success(t.apply.success.title);
   297→    } catch {
   298→      toast.error(t.apply.errors.submitFailed);
   299→    } finally {
   300→      setSubmitting(false);
   301→    }
   302→  }
   303→
   304→  function resetForm() {
   305→    submittedRef.current = false;
   306→    draftDismissedRef.current = false;
   307→    setValues(INITIAL_VALUES);
   308→    setErrors({});
   309→    setStep(0);
   310→    setSuccess(null);
   311→    setCvFile(null);
   312→    setIntroFile(null);
   313→    setCvError(null);
   314→    setIntroError(null);
   315→    onPositionIdChange("");
   316→    try {
   317→      window.localStorage.removeItem(DRAFT_KEY);
   318→    } catch {
   319→      // abaikan
   320→    }
   321→  }
   322→
   323→  function restoreDraft() {
   324→    if (!draft) return;
   325→    const restored: FormValues = { ...INITIAL_VALUES };
   326→    for (const key of VALUE_KEYS) {
   327→      const raw = draft.values?.[key];
   328→      if (typeof raw === "string") restored[key] = raw;
   329→    }
   330→    setValues(restored);
   331→    draftDismissedRef.current = false;
   332→    if (
   333→      typeof draft.positionId === "string" &&
   334→      positions.some((p) => p.id === draft.positionId)
   335→    ) {
   336→      onPositionIdChange(draft.positionId);
   337→    }
   338→    setDraft(null);
   339→    toast.success(t.apply.draft.title);
   340→  }
   341→
   342→  function discardDraft() {
   343→    draftDismissedRef.current = true;
   344→    try {
   345→      window.localStorage.removeItem(DRAFT_KEY);
   346→    } catch {
   347→      // abaikan
   348→    }
   349→    setDraft(null);
   350→  }
   351→
   352→  async function copyTrackingCode() {
   353→    if (!success?.trackingCode) return;
   354→    try {
   355→      await navigator.clipboard.writeText(success.trackingCode);
   356→      toast.success(t.apply.success.copied);
   357→    } catch {
   358→      toast.error(t.apply.errors.submitFailed);
   359→    }
   360→  }
   361→
   362→  function goToStatus() {
   363→    document.getElementById("status")?.scrollIntoView({ behavior: "smooth" });
   364→  }
   365→
   366→  if (success) {
   367→    return (
   368→      <div
   369→        role="status"
   370→        className="flex flex-col items-center justify-center gap-4 py-8 text-center"
   371→      >
   372→        <div className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-500/15">
   373→          <CheckCircle2
   374→            className="h-12 w-12 text-emerald-600 dark:text-emerald-400"
   375→            aria-hidden="true"
   376→          />
   377→        </div>
   378→        <h3 className="text-2xl font-bold">{t.apply.success.title}</h3>
   379→        <p className="text-sm font-medium">
   380→          {t.apply.success.thanksTo} {success.name}
   381→        </p>
   382→        <p className="max-w-md text-sm text-muted-foreground">{t.apply.success.body}</p>
   383→
   384→        <div className="mt-2 flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border bg-muted/40 p-5">
   385→          <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   386→            {t.apply.success.trackingLabel}
   387→          </p>
   388→          <div className="flex w-full items-center justify-center gap-3">
   389→            <code className="rounded-lg bg-background px-4 py-2 font-mono text-xl tracking-widest sm:text-2xl">
   390→              {success.trackingCode || "-"}
   391→            </code>
   392→            <Button
   393→              variant="outline"
   394→              size="icon"
   395→              className="h-10 w-10 shrink-0"
   396→              aria-label={t.apply.success.copy}
   397→              onClick={() => void copyTrackingCode()}
   398→            >
   399→              <Copy className="h-4 w-4" aria-hidden="true" />
   400→            </Button>
   401→          </div>
   402→          <p className="text-xs text-muted-foreground">{t.apply.success.saveNote}</p>
   403→        </div>
   404→
   405→        <div className="mt-2 flex flex-col gap-3 sm:flex-row">
   406→          <Button onClick={goToStatus}>{t.apply.success.checkStatus}</Button>
   407→          <Button variant="outline" onClick={resetForm}>
   408→            {t.apply.success.another}
   409→          </Button>
   410→        </div>
   411→      </div>
   412→    );
   413→  }
   414→
   415→  const stepLabels = t.apply.steps;
   416→  const selectedPosition = positions.find((p) => p.id === positionId);
   417→
   418→  return (
   419→    <div className="flex flex-col gap-6">
   420→      <div>
   421→        <h3 className="text-lg font-semibold">{t.apply.formTitle}</h3>
   422→        <p className="mt-1 text-sm text-muted-foreground">
   423→          {t.apply.requiredBefore} <span className="text-rose-600">*</span>{" "}
   424→          {t.apply.requiredAfter}
   425→        </p>
   426→      </div>
   427→
   428→      {draft ? (
   429→        <Card className="flex-row items-center justify-between gap-3 rounded-xl border-amber-200 bg-amber-50/60 p-4 dark:border-amber-500/30 dark:bg-amber-500/10">
   430→          <div className="min-w-0">
   431→            <p className="text-sm font-semibold">{t.apply.draft.title}</p>
   432→            <p className="truncate text-xs text-muted-foreground">
   433→              {t.apply.draft.body}
   434→              {typeof draft.savedAt === "number"
   435→                ? ` (${t.apply.draft.savedPrefix} ${new Date(draft.savedAt).toLocaleString("id-ID")})`
   436→                : ""}
   437→            </p>
   438→          </div>
   439→          <div className="flex shrink-0 items-center gap-2">
   440→            <Button size="sm" variant="outline" onClick={discardDraft}>
   441→              {t.apply.draft.discard}
   442→            </Button>
   443→            <Button size="sm" onClick={restoreDraft}>
   444→              {t.apply.draft.restore}
   445→            </Button>
   446→          </div>
   447→        </Card>
   448→      ) : null}
   449→
   450→      {/* Stepper */}
   451→      <ol className="flex items-center gap-2" aria-label={t.apply.stepOf}>
   452→        {stepLabels.map((label, index) => {
   453→          const isDone = index < step;
   454→          const isActive = index === step;
   455→          return (
   456→            <li key={label} className="flex flex-1 items-center gap-2 last:flex-none">
   457→              <div className="flex items-center gap-2">
   458→                <span
   459→                  aria-current={isActive ? "step" : undefined}
   460→                  className={cn(
   461→                    "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border text-xs font-bold transition-colors",
   462→                    isDone &&
   463→                      "border-emerald-500 bg-emerald-500 text-white dark:border-emerald-400 dark:bg-emerald-400 dark:text-emerald-950",
   464→                    isActive && "border-primary ring-primary ring-offset-background ring-2",
   465→                    !isDone && !isActive && "border-border text-muted-foreground",
   466→                  )}
   467→                >
   468→                  {isDone ? (
   469→                    <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   470→                  ) : (
   471→                    index + 1
   472→                  )}
   473→                </span>
   474→                <span
   475→                  className={cn(
   476→                    "hidden text-xs font-medium sm:block",
   477→                    isActive ? "text-foreground" : "text-muted-foreground",
   478→                  )}
   479→                >
   480→                  {label}
   481→                </span>
   482→              </div>
   483→              {index < stepLabels.length - 1 ? (
   484→                <span
   485→                  aria-hidden="true"
   486→                  className={cn(
   487→                    "h-px flex-1",
   488→                    index < step ? "bg-emerald-500 dark:bg-emerald-400" : "bg-border",
   489→                  )}
   490→                />
   491→              ) : null}
   492→            </li>
   493→          );
   494→        })}
   495→      </ol>
   496→
   497→      <form onSubmit={handleSubmit} noValidate className="flex flex-col gap-5">
   498→        {/* LANGKAH 1: Data Diri */}
   499→        {step === 0 ? (
   500→          <div className="flex flex-col gap-5">
   501→            <div className="grid gap-5 sm:grid-cols-2">
   502→              <div className="flex flex-col gap-2">
   503→                <Label htmlFor="apply-name">
   504→                  {t.apply.fields.name} <span className="text-rose-600">*</span>
   505→                </Label>
   506→                <Input
   507→                  id="apply-name"
   508→                  name="name"
   509→                  value={values.name}
   510→                  onChange={(e) => setField("name", e.target.value)}
   511→                  placeholder={t.apply.fields.namePh}
   512→                  autoComplete="name"
   513→                  className="h-11"
   514→                  aria-invalid={errors.name ? true : undefined}
   515→                  aria-describedby={errors.name ? "apply-name-error" : undefined}
   516→                />
   517→                {errors.name ? (
   518→                  <p id="apply-name-error" className="text-sm text-rose-600">
   519→                    {errors.name}
   520→                  </p>
   521→                ) : null}
   522→              </div>
   523→
   524→              <div className="flex flex-col gap-2">
   525→                <Label htmlFor="apply-email">
   526→                  {t.apply.fields.email} <span className="text-rose-600">*</span>
   527→                </Label>
   528→                <Input
   529→                  id="apply-email"
   530→                  name="email"
   531→                  type="email"
   532→                  value={values.email}
   533→                  onChange={(e) => setField("email", e.target.value)}
   534→                  placeholder={t.apply.fields.emailPh}
   535→                  autoComplete="email"
   536→                  className="h-11"
   537→                  aria-invalid={errors.email ? true : undefined}
   538→                  aria-describedby={errors.email ? "apply-email-error" : undefined}
   539→                />
   540→                {errors.email ? (
   541→                  <p id="apply-email-error" className="text-sm text-rose-600">
   542→                    {errors.email}
   543→                  </p>
   544→                ) : null}
   545→              </div>
   546→            </div>
   547→
   548→            <div className="grid gap-5 sm:grid-cols-2">
   549→              <div className="flex flex-col gap-2">
   550→                <Label htmlFor="apply-phone">
   551→                  {t.apply.fields.phone} <span className="text-rose-600">*</span>
   552→                </Label>
   553→                <Input
   554→                  id="apply-phone"
   555→                  name="phone"
   556→                  type="tel"
   557→                  inputMode="tel"
   558→                  value={values.phone}
   559→                  onChange={(e) => setField("phone", e.target.value)}
   560→                  placeholder={t.apply.fields.phonePh}
   561→                  autoComplete="tel"
   562→                  className="h-11"
   563→                  aria-invalid={errors.phone ? true : undefined}
   564→                  aria-describedby={errors.phone ? "apply-phone-error" : undefined}
   565→                />
   566→                {errors.phone ? (
   567→                  <p id="apply-phone-error" className="text-sm text-rose-600">
   568→                    {errors.phone}
   569→                  </p>
   570→                ) : null}
   571→              </div>
   572→
   573→              <div className="flex flex-col gap-2">
   574→                <Label htmlFor="apply-position">
   575→                  {t.apply.fields.position} <span className="text-rose-600">*</span>
   576→                </Label>
   577→                <Select
   578→                  value={positionId || undefined}
   579→                  onValueChange={(value) => {
