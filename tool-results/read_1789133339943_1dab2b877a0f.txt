    93→function moveItem<T>(items: T[], index: number, direction: -1 | 1): T[] {
    94→  const next = [...items];
    95→  const target = index + direction;
    96→  if (target < 0 || target >= next.length) return next;
    97→  const [item] = next.splice(index, 1);
    98→  next.splice(target, 0, item);
    99→  return next;
   100→}
