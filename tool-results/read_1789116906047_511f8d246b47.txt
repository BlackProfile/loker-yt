   814→/** JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions). */
   815→function useJobPostingJsonLd(positions: Position[], siteName: string) {
   816→  useEffect(() => {
   817→    const jobPostings = positions.map((position) => ({
   818→      "@context": "https://schema.org",
   819→      "@type": "JobPosting",
   820→      title: position.title,
   821→      description: position.description,
   822→      hiringOrganization: { "@type": "Organization", name: siteName },
   823→      jobLocationType: "TELECOMMUTE",
   824→      employmentType: employmentTypeOf(position.type),
   825→      datePosted: position.createdAt,
   826→      ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   827→    }));
   828→    const script = document.createElement("script");
   829→    script.id = "jobposting-ld";
   830→    script.type = "application/ld+json";
   831→    script.textContent = JSON.stringify(jobPostings);
   832→    document.head.appendChild(script);
   833→    return () => {
   834→      document.getElementById("jobposting-ld")?.remove();
   835→    };
   836→  }, [positions, siteName]);
   837→}
