   137→  "email",

   141→  "experience",

   146→  values?: Partial<Record<keyof FormValues, unknown>>;
   147→  positionId?: unknown;

   157→export function ApplyWizard({
   158→  positions,
   159→  positionId,
   160→  onPositionIdChange,
   161→}: ApplyWizardProps) {
   162→  const { t } = useLang();
   163→  const [values, setValues] = useState<FormValues>(INITIAL_VALUES);
   164→  const [errors, setErrors] = useState<FormErrors>({});
   165→  const [step, setStep] = useState(0);
   166→  const [submitting, setSubmitting] = useState(false);
   167→  // Pratinjau & konfirmasi sebelum pengiriman (tidak ada kirim otomatis).
   168→  const [confirmOpen, setConfirmOpen] = useState(false);
   169→  const [agreed, setAgreed] = useState(false);
   170→  const [direction, setDirection] = useState(1);
   171→  const [success, setSuccess] = useState<{ name: string; trackingCode: string } | null>(
   172→    null,
   173→  );
   174→
   175→  // File unggahan (tidak masuk draft).
   176→  const [cvFile, setCvFile] = useState<File | null>(null);
   177→  const [introFile, setIntroFile] = useState<File | null>(null);
   178→  const [cvError, setCvError] = useState<string | null>(null);
   179→  const [introError, setIntroError] = useState<string | null>(null);
   180→  const [cvDragging, setCvDragging] = useState(false);
   181→  const [introDragging, setIntroDragging] = useState(false);
   182→
   183→  // Draft autosave.
   184→  const [draft, setDraft] = useState<StoredDraft | null>(null);
   185→  const submittedRef = useRef(false);
   186→  const draftDismissedRef = useRef(false);
   187→
   188→  useEffect(() => {
   189→    try {
   190→      const raw = window.localStorage.getItem(DRAFT_KEY);
   191→      if (!raw) return;
   192→      const parsed = JSON.parse(raw) as StoredDraft;
   193→      if (parsed && typeof parsed === "object" && parsed.values) setDraft(parsed);
   194→    } catch {
   195→      // draft rusak -> abaikan
   196→    }
   197→  }, []);
   198→
   199→  useEffect(() => {
   200→    if (submittedRef.current || draftDismissedRef.current) return;
   201→    const hasContent =
   202→      VALUE_KEYS.some((key) => values[key].trim().length > 0) || positionId;
   203→    if (!hasContent) return;
   204→    const timer = window.setTimeout(() => {
   205→      try {
   206→        window.localStorage.setItem(
   207→          DRAFT_KEY,
   208→          JSON.stringify({ values, positionId, savedAt: Date.now() }),
   209→        );
   210→      } catch {
   211→        // penyimpanan penuh / tidak tersedia
   212→      }
   213→    }, AUTOSAVE_DELAY_MS);
   214→    return () => window.clearTimeout(timer);
   215→  }, [values, positionId]);
   216→
   217→  const setField = (key: keyof FormValues, value: string) => {
   218→    setValues((prev) => ({ ...prev, [key]: value }));
   219→    setErrors((prev) => (prev[key] ? { ...prev, [key]: undefined } : prev));
   220→  };
   221→
   222→  const clearPositionError = () => {
   223→    setErrors((prev) =>
   224→      prev.positionId ? { ...prev, positionId: undefined } : prev,
   225→    );
   226→  };
   227→
   228→  function validateStep1(): FormErrors {
   229→    const next: FormErrors = {};
   230→    if (!values.name.trim()) next.name = t.apply.errors.name;
   231→    if (!values.email.trim()) next.email = t.apply.errors.emailRequired;
   232→    else if (!EMAIL_RE.test(values.email.trim())) next.email = t.apply.errors.emailInvalid;
   233→    if (!values.phone.trim()) next.phone = t.apply.errors.phoneRequired;
   234→    else if (values.phone.replace(/\D/g, "").length < 8)
   235→      next.phone = t.apply.errors.phoneMin;
   236→    if (positions.length > 0 && !positionId) next.positionId = t.apply.errors.position;
   237→    return next;
   238→  }
   239→
   240→  function validateStep2(): FormErrors {
   241→    const next: FormErrors = {};
   242→    if (values.experience.trim().length < MIN_TEXT_LENGTH)
   243→      next.experience = t.apply.errors.experience;
   244→    if (values.motivation.trim().length < MIN_TEXT_LENGTH)
   245→      next.motivation = t.apply.errors.motivation;
   246→    return next;
   247→  }
   248→
   249→  function goToStep(next: number) {
   250→    setDirection(next > step ? 1 : -1);
   251→    setStep(next);
   252→  }
   253→
   254→  function goNext() {
   255→    const next = step === 0 ? validateStep1() : validateStep2();
   256→    setErrors((prev) => ({ ...prev, ...next }));
   257→    if (Object.values(next).some(Boolean)) return;
   258→    goToStep(Math.min(2, step + 1));
   259→  }
   260→
   261→  function goBack() {
   262→    goToStep(Math.max(0, step - 1));
   263→  }
   264→
   265→  /** Validasi seluruh formulir; bila ada galat, lompat ke langkah pertama yang bermasalah. */
   266→  function validateAllAndJump(): boolean {
   267→    const e1 = validateStep1();
   268→    const e2 = validateStep2();
   269→    setErrors((prev) => ({ ...prev, ...e1, ...e2 }));
   270→    if (e1.name || e1.email || e1.phone || e1.positionId) {
   271→      goToStep(0);
   272→      return false;
   273→    }
   274→    if (Object.values({ ...e1, ...e2 }).some(Boolean)) {
   275→      goToStep(1);
   276→      return false;
   277→    }
   278→    return true;
   279→  }
   280→
   281→  function acceptCv(file: File | null) {
   282→    if (!file) return;
   283→    if (file.type !== "application/pdf") {
   284→      setCvError(t.apply.errors.cvType);
   285→      toast.error(t.apply.errors.cvType);
   286→      return;
   287→    }
   288→    if (file.size > CV_MAX_BYTES) {
   289→      setCvError(t.apply.errors.cvSize);
   290→      toast.error(t.apply.errors.cvSize);
   291→      return;
   292→    }
   293→    setCvFile(file);
   294→    setCvError(null);
   295→  }
   296→
   297→  function acceptIntro(file: File | null) {
   298→    if (!file) return;
   299→    const typeOk =
   300→      file.type.startsWith("audio/") || /\.(mp3|wav|m4a)$/i.test(file.name);
   301→    if (!typeOk) {
   302→      setIntroError(t.apply.errors.introType);
   303→      toast.error(t.apply.errors.introType);
   304→      return;
   305→    }
   306→    if (file.size > INTRO_MAX_BYTES) {
   307→      setIntroError(t.apply.errors.introSize);
   308→      toast.error(t.apply.errors.introSize);
   309→      return;
   310→    }
   311→    setIntroFile(file);
   312→    setIntroError(null);
   313→  }
   314→
   315→  function onCvInput(event: ChangeEvent<HTMLInputElement>) {
   316→    acceptCv(event.target.files?.[0] ?? null);
   317→    event.target.value = "";
   318→  }
   319→
   320→  function onIntroInput(event: ChangeEvent<HTMLInputElement>) {
   321→    acceptIntro(event.target.files?.[0] ?? null);
   322→    event.target.value = "";
   323→  }
   324→
   325→  function onDropFactory(
   326→    acceptor: (file: File | null) => void,
   327→    setDragging: (value: boolean) => void,
   328→  ) {
   329→    return (event: DragEvent<HTMLLabelElement>) => {
   330→      event.preventDefault();
   331→      setDragging(false);
   332→      acceptor(event.dataTransfer.files?.[0] ?? null);
   333→    };
   334→  }
   335→
   336→  /**
   337→   * Pengiriman sesungguhnya. Hanya dipanggil setelah pengguna melihat pratinjau
   338→   * dan menekan konfirmasi pada dialog — tidak pernah kirim otomatis.
   339→   */
   340→  async function doSubmit() {
   341→    setSubmitting(true);
   342→    try {
   343→      const fd = new FormData();
   344→      fd.append("name", values.name.trim());
   345→      fd.append("email", values.email.trim());
   346→      fd.append("phone", values.phone.trim());
   347→      if (positionId) fd.append("positionId", positionId);
   348→      if (values.portfolioUrl.trim())
   349→        fd.append("portfolioUrl", values.portfolioUrl.trim());
   350→      if (values.socialLinks.trim())
   351→        fd.append("socialLinks", values.socialLinks.trim());
   352→      fd.append("experience", values.experience.trim());
   353→      fd.append("motivation", values.motivation.trim());
   354→      if (cvFile) fd.append("cvFile", cvFile);
   355→      if (introFile) fd.append("introFile", introFile);
   356→
   357→      const res = await fetch("/api/applications", { method: "POST", body: fd });
   358→      const data: unknown = await res.json().catch(() => null);
   359→      if (!res.ok) {
   360→        const serverError =
   361→          typeof data === "object" &&
   362→          data !== null &&
   363→          "error" in data &&
   364→          typeof (data as { error: unknown }).error === "string"
   365→            ? (data as { error: string }).error
   366→            : null;
   367→        toast.error(serverError ?? t.apply.errors.submitFailed);
   368→        return;
   369→      }
   370→      const trackingCode =
   371→        typeof data === "object" &&
   372→        data !== null &&
   373→        "trackingCode" in data &&
   374→        typeof (data as { trackingCode: unknown }).trackingCode === "string"
   375→          ? (data as { trackingCode: string }).trackingCode
   376→          : "";
   377→
   378→      submittedRef.current = true;
   379→      try {
   380→        window.localStorage.removeItem(DRAFT_KEY);
   381→      } catch {
   382→        // abaikan
   383→      }
   384→      setSuccess({ name: values.name.trim(), trackingCode });
   385→      toast.success(t.apply.success.title);
   386→    } catch {
   387→      toast.error(t.apply.errors.submitFailed);
   388→    } finally {
   389→      setSubmitting(false);
   390→    }
   391→  }
   392→
   393→  function handleSubmit(event: FormEvent<HTMLFormElement>) {
   394→    event.preventDefault();
   395→    if (submitting) return;
   396→
   397→    // Langkah 0-1: validasi per langkah lalu maju.
   398→    if (step === 0 || step === 1) {
   399→      goNext();
   400→      return;
   401→    }
   402→
   403→    // Langkah 2 (Berkas): validasi semuanya, lalu tampilkan pratinjau.
   404→    if (step === 2) {
   405→      if (validateAllAndJump()) goToStep(3);
   406→      return;
   407→    }
   408→
   409→    // Langkah 3 (Pratinjau): wajib pernyataan kebenaran data sebelum konfirmasi.
   410→    if (!agreed) {
   411→      toast.error(t.apply.preview.agreeRequired);
   412→      return;
   413→    }
   414→    setConfirmOpen(true);
   415→  }
   416→
   417→  function resetForm() {
   418→    submittedRef.current = false;
   419→    draftDismissedRef.current = false;
   420→    setValues(INITIAL_VALUES);
   421→    setErrors({});
   422→    setStep(0);
   423→    setSuccess(null);
   424→    setCvFile(null);
   425→    setIntroFile(null);
   426→    setCvError(null);
   427→    setIntroError(null);
   428→    setAgreed(false);
   429→    setConfirmOpen(false);
   430→    setDirection(1);
   431→    onPositionIdChange("");
   432→    try {
   433→      window.localStorage.removeItem(DRAFT_KEY);
   434→    } catch {
   435→      // abaikan
   436→    }
   437→  }
   438→
   439→  function restoreDraft() {
   440→    if (!draft) return;
   441→    const restored: FormValues = { ...INITIAL_VALUES };
   442→    for (const key of VALUE_KEYS) {
   443→      const raw = draft.values?.[key];
   444→      if (typeof raw === "string") restored[key] = raw;
   445→    }
   446→    setValues(restored);
   447→    draftDismissedRef.current = false;
   448→    if (
   449→      typeof draft.positionId === "string" &&
   450→      positions.some((p) => p.id === draft.positionId)
   451→    ) {
   452→      onPositionIdChange(draft.positionId);
   453→    }
   454→    setDraft(null);
   455→    toast.success(t.apply.draft.title);
   456→  }
   457→
   458→  function discardDraft() {
   459→    draftDismissedRef.current = true;
   460→    try {
   461→      window.localStorage.removeItem(DRAFT_KEY);
   462→    } catch {
   463→      // abaikan
   464→    }
   465→    setDraft(null);
   466→  }
   467→
   468→  async function copyTrackingCode() {
   469→    if (!success?.trackingCode) return;
   470→    try {
   471→      await navigator.clipboard.writeText(success.trackingCode);
   472→      toast.success(t.apply.success.copied);
   473→    } catch {
   474→      toast.error(t.apply.errors.submitFailed);
   475→    }
   476→  }
   477→
   478→  function goToStatus() {
   479→    document.getElementById("status")?.scrollIntoView({ behavior: "smooth" });
   480→  }
   481→
   482→  if (success) {
   483→    return (
   484→      <motion.div
   485→        role="status"
   486→        initial={{ opacity: 0, y: 16 }}
   487→        animate={{ opacity: 1, y: 0 }}
   488→        transition={{ duration: 0.35, ease: "easeOut" }}
   489→        className="flex flex-col items-center justify-center gap-4 py-8 text-center"
   490→      >
   491→        <motion.div
   492→          initial={{ scale: 0.4, opacity: 0 }}
   493→          animate={{ scale: 1, opacity: 1 }}
   494→          transition={{ type: "spring", stiffness: 260, damping: 16, delay: 0.1 }}
   495→          className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-500/15"
   496→        >
   497→          <CheckCircle2
   498→            className="h-12 w-12 text-emerald-600 dark:text-emerald-400"
   499→            aria-hidden="true"
   500→          />
   501→        </motion.div>
   502→        <h3 className="text-2xl font-bold">{t.apply.success.title}</h3>
   503→        <p className="text-sm font-medium">
   504→          {t.apply.success.thanksTo} {success.name}
   505→        </p>
   506→        <p className="max-w-md text-sm text-muted-foreground">{t.apply.success.body}</p>
   507→
   508→        <div className="mt-2 flex w-full max-w-md flex-col items-center gap-3 rounded-2xl border bg-muted/40 p-5">
   509→          <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   510→            {t.apply.success.trackingLabel}
   511→          </p>
   512→          <div className="flex w-full items-center justify-center gap-3">
   513→            <code className="rounded-lg bg-background px-4 py-2 font-mono text-xl tracking-widest sm:text-2xl">
   514→              {success.trackingCode || "-"}
   515→            </code>
   516→            <Button
   517→              variant="outline"
   518→              size="icon"
   519→              className="h-10 w-10 shrink-0"
   520→              aria-label={t.apply.success.copy}
   521→              onClick={() => void copyTrackingCode()}
   522→            >
   523→              <Copy className="h-4 w-4" aria-hidden="true" />
   524→            </Button>
   525→          </div>
   526→          <p className="text-xs text-muted-foreground">{t.apply.success.saveNote}</p>
   527→        </div>
   528→
   529→        <div className="mt-2 flex flex-col gap-3 sm:flex-row">
   530→          <Button onClick={goToStatus}>{t.apply.success.checkStatus}</Button>
   531→          <Button variant="outline" onClick={resetForm}>
   532→            {t.apply.success.another}
   533→          </Button>
   534→        </div>
   535→      </motion.div>
   536→    );
   537→  }
   538→
   539→  const stepLabels = t.apply.steps;
   540→  const selectedPosition = positions.find((p) => p.id === positionId);
   541→
   542→  return (
   543→    <div className="flex flex-col gap-6">
   544→      <div>
   545→        <h3 className="text-lg font-semibold">{t.apply.formTitle}</h3>
   546→        <p className="mt-1 text-sm text-muted-foreground">
   547→          {t.apply.requiredBefore} <span className="text-rose-600">*</span>{" "}
   548→          {t.apply.requiredAfter}
   549→        </p>
   550→      </div>
   551→
   552→      {draft ? (
   553→        <Card className="flex-row items-center justify-between gap-3 rounded-xl border-amber-200 bg-amber-50/60 p-4 dark:border-amber-500/30 dark:bg-amber-500/10">
   554→          <div className="min-w-0">
   555→            <p className="text-sm font-semibold">{t.apply.draft.title}</p>
   556→            <p className="truncate text-xs text-muted-foreground">
   557→              {t.apply.draft.body}
   558→              {typeof draft.savedAt === "number"
   559→                ? ` (${t.apply.draft.savedPrefix} ${new Date(draft.savedAt).toLocaleString("id-ID")})`
   560→                : ""}
   561→            </p>
   562→          </div>
   563→          <div className="flex shrink-0 items-center gap-2">
   564→            <Button size="sm" variant="outline" onClick={discardDraft}>
   565→              {t.apply.draft.discard}
   566→            </Button>
   567→            <Button size="sm" onClick={restoreDraft}>
   568→              {t.apply.draft.restore}
   569→            </Button>
   570→          </div>
   571→        </Card>
   572→      ) : null}
   573→
   574→      {/* Stepper */}
   575→      <ol className="flex items-center gap-2" aria-label={t.apply.stepOf}>
   576→        {stepLabels.map((label, index) => {
   577→          const isDone = index < step;
   578→          const isActive = index === step;
   579→          return (
   580→            <li key={label} className="flex flex-1 items-center gap-2 last:flex-none">
   581→              <div className="flex items-center gap-2">
   582→                <span
