   206→function formPublicationMode(form: FormState): {
   207→  key: keyof typeof PUB_MODE;
   208→  hint: string;
   209→} {
   210→  if (!form.isActive) {
   211→    return { key: "draft", hint: "Posisi tidak tampil di halaman publik." };
   212→  }
   213→  if (form.publishAtLocal) {
   214→    const t = new Date(form.publishAtLocal).getTime();
   215→    if (!Number.isNaN(t) && t > Date.now()) {
   216→      return {
   217→        key: "terjadwal",
   218→        hint: "Posisi otomatis tayang tepat pada jadwal publikasi di atas.",
   219→      };
   220→    }
   221→  }
   222→  return { key: "tayang", hint: "Posisi tampil di halaman publik sekarang." };
   223→}

   244→export function PositionFormDialog({
   245→  open,
   246→  onOpenChange,
   247→  editing,
   248→  statsRow,
   249→  onSaved,
   250→}: {
   251→  open: boolean;
   252→  onOpenChange: (open: boolean) => void;
   253→  editing: Position | null;
   254→  statsRow: PositionStatsRow | null;
   255→  onSaved: () => void;
   256→}) {
   257→  const { reportError } = useAdminSession();

   259→    editing ? buildFormState(editing) : EMPTY_FORM
   260→  );
   261→  const [validationErrors, setValidationErrors] = useState<string[]>([]);
   262→  const [saving, setSaving] = useState(false);
   263→  const [uploadingCover, setUploadingCover] = useState(false);
   264→  const [aiCoverLoading, setAiCoverLoading] = useState(false);
   265→  const fileInputRef = useRef<HTMLInputElement>(null);
   266→
   267→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) =>
   268→    setForm((f) => ({ ...f, [key]: value }));
   269→
   270→  const cleanedStages = useMemo(
   271→    () => form.stages.map((s) => s.trim()).filter(Boolean),
   272→    [form.stages]
   273→  );
   274→
   275→  // Lamaran pada tahap di luar daftar pipeline tersimpan (dari stats funnel).
   276→  const outOfStageApps = useMemo(() => {
   277→    if (!editing || !statsRow) return 0;
   278→    const inFunnel = statsRow.funnel.reduce((sum, f) => sum + f.count, 0);
   279→    return Math.max(0, statsRow.applications - inFunnel);
   280→  }, [editing, statsRow]);
   281→
   282→  const pubMode = formPublicationMode(form);
   283→  const pub = PUB_MODE[pubMode.key];
   284→
   285→  function validate(): string[] {
   286→    const errors: string[] = [];
   287→    if (form.title.trim().length < 3)
   288→      errors.push("Judul posisi minimal 3 karakter.");
   289→    if (!form.department.trim()) errors.push("Departemen wajib diisi.");
   290→    if (form.description.trim().length < 10)
   291→      errors.push("Deskripsi minimal 10 karakter.");
   292→    if (form.salaryText.trim().length > 80)
   293→      errors.push("Teks gaji maksimal 80 karakter.");
   294→    if (
   295→      form.examples.some(
   296→        (url) => url.trim().length > 0 && !/^https?:\/\//i.test(url.trim())
   297→      )
   298→    )
   299→      errors.push("Contoh karya harus diawali http:// atau https://.");
   300→    if (
   301→      form.assignmentUrl.trim().length > 0 &&
   302→      !/^https?:\/\//i.test(form.assignmentUrl.trim())
   303→    )
   304→      errors.push("URL tes/assignment harus diawali http:// atau https://.");
   305→    if (form.screeningQuestions.some((q) => q.label.trim().length < 3))
   306→      errors.push("Label pertanyaan screening minimal 3 karakter.");
   307→    if (cleanedStages.length > 12)
   308→      errors.push("Pipeline tahap maksimal 12 tahap.");
   309→    if (form.maxApplicants.trim() !== "") {
   310→      if (!isInt(form.maxApplicants) || Number(form.maxApplicants) < 1 || Number(form.maxApplicants) > 10000)
   311→        errors.push("Kuota pelamar harus angka bulat 1-10000, atau dikosongkan.");
   312→    }
   313→    if (form.autoShortlistScore.trim() !== "") {
   314→      if (!isInt(form.autoShortlistScore) || Number(form.autoShortlistScore) < 0 || Number(form.autoShortlistScore) > 100)
   315→        errors.push("Skor auto-shortlist harus angka bulat 0-100, atau dikosongkan.");
   316→    }
   317→    if (form.order.trim() !== "" && !isInt(form.order))
   318→      errors.push("Urutan harus berupa bilangan bulat.");
   319→    return errors;
   320→  }
   321→
   322→  async function handleSave() {
   323→    if (saving) return;
   324→    const errors = validate();
   325→    setValidationErrors(errors);
   326→    if (errors.length > 0) {
   327→      toast.error(errors[0]);
   328→      return;
   329→    }
   330→    setSaving(true);
   331→    const payload: Record<string, unknown> = {
   332→      title: form.title.trim(),
   333→      department: form.department.trim(),
   334→      type: form.type,
   335→      location: form.location.trim() || "Remote",
   336→      description: form.description.trim(),
   337→      requirements: form.requirements.map((r) => r.trim()).filter(Boolean),
   338→      isActive: form.isActive,
   339→      publishAt: localInputToIso(form.publishAtLocal),
   340→      closesAt: localInputToIso(form.closesAtLocal),
   341→      coverFileId: form.coverFileId,
   342→      salaryText: form.salaryText.trim() || null,
   343→      salaryVisible: form.salaryVisible,
   344→      benefits: form.benefits.map((b) => b.trim()).filter(Boolean),
   345→      examples: form.examples.map((e) => e.trim()).filter(Boolean),
   346→      urgent: form.urgent,
   347→      featured: form.featured,
   348→      requireCv: form.requireCv,
   349→      requireIntro: form.requireIntro,
   350→      requirePortfolio: form.requirePortfolio,
   351→      maxApplicants:
   352→        form.maxApplicants.trim() === "" ? null : Number(form.maxApplicants),
   353→      screeningQuestions: form.screeningQuestions.map((q) => ({
   354→        id: q.id,
   355→        label: q.label.trim(),
   356→        required: q.required,
   357→      })),
   358→      stages: cleanedStages,
   359→      aiCriteria: form.aiCriteria.trim() || null,
   360→      autoShortlistScore:
   361→        form.autoShortlistScore.trim() === ""
   362→          ? null
   363→          : Number(form.autoShortlistScore),
   364→      autoShortlistStage:
   365→        cleanedStages.length === 0 ? null : form.autoShortlistStage || null,
   366→      applyTemplate: form.applyTemplate.trim() || null,
   367→      acceptTemplate: form.acceptTemplate.trim() || null,
   368→      rejectTemplate: form.rejectTemplate.trim() || null,
   369→      assignmentTitle: form.assignmentTitle.trim() || null,
   370→      assignmentUrl: form.assignmentUrl.trim() || null,
   371→      assignmentNote: form.assignmentNote.trim() || null,
   372→      rubricCriteria: form.rubricCriteria.map((r) => r.trim()).filter(Boolean),
   373→      checklistTemplate: form.checklistTemplate.map((c) => c.trim()).filter(Boolean),
   374→      noteTemplates: form.noteTemplates.map((n) => n.trim()).filter(Boolean),
   375→    };
   376→    if (form.order.trim() !== "") payload.order = Number(form.order);
   377→
   378→    try {
   379→      if (editing) {
   380→        await apiPatch<Position>(`/api/admin/positions/${editing.id}`, payload);
   381→        toast.success("Posisi diperbarui");
   382→      } else {
   383→        await apiPost<Position>("/api/admin/positions", payload);
   384→        toast.success("Posisi ditambahkan");
   385→      }
   386→      onOpenChange(false);
   387→      onSaved();
   388→    } catch (err) {
   389→      reportError(err);
   390→    } finally {
   391→      setSaving(false);
   392→    }
   393→  }

   395→  function handleUploadCover(file: File) {
   396→    if (!COVER_MIME.includes(file.type)) {
   397→      toast.error("Format gambar harus PNG, JPEG, atau WebP.");
   398→      return;
   399→    }
   400→    if (file.size > COVER_MAX_BYTES) {
   401→      toast.error("Ukuran gambar maksimal 3 MB.");
   402→      return;
   403→    }
   404→    setUploadingCover(true);
   405→    const fd = new FormData();
   406→    fd.append("file", file);
   407→    apiFetch<AdminUploadResponse>("/api/admin/upload", {
   408→      method: "POST",
   409→      body: fd,
   410→    })
   411→      .then((res) => {
   412→        set("coverFileId", res.fileId);
   413→        toast.success("Cover berhasil diunggah");
   414→      })
   415→      .catch((err) => reportError(err))
   416→      .finally(() => {
   417→        setUploadingCover(false);
   418→        if (fileInputRef.current) fileInputRef.current.value = "";
   419→      });
   420→  }

   422→  function handleAiCover() {
   423→    if (!editing || aiCoverLoading) return;
   424→    setAiCoverLoading(true);
   425→    const tid = toast.loading(
   426→      "Membuat cover dengan AI... proses bisa memakan waktu hingga 1 menit."
   427→    );
   428→    apiPost<AiCoverResponse>("/api/admin/ai/cover", { positionId: editing.id })
   429→      .then((res) => {
   430→        if (res.ok) {
   431→          set("coverFileId", res.fileId);
   432→          toast.success("Cover AI berhasil dibuat", { id: tid });
   433→          onSaved();
   434→        } else {
   435→          toast.error(res.error, { id: tid });
   436→        }
   437→      })
   438→      .catch((err) => {
   439→        reportError(err);
   440→        toast.dismiss(tid);
   441→      })
   442→      .finally(() => setAiCoverLoading(false));
   443→  }

   451→            {editing
   452→              ? "Perbarui semua pengaturan posisi lowongan."
   453→              : "Tambahkan posisi lowongan baru untuk halaman publik."}
   454→          </DialogDescription>
   455→        </DialogHeader>
