   280→  function validateStep1(): FormErrors {

   287→      next.phone = t.apply.errors.phoneMin;

   313→      next.portfolioUrl = t.apply.errors.portfolioRequired;
   314→    }
   315→    return next;
   316→  }

   324→  /** Error screening pertama dari peta error (untuk scroll + toast). */
   325→  function firstScreeningErrorOf(next: FormErrors): { id: string; message: string } | null {
   326→    for (const [key, message] of Object.entries(next)) {
   327→      if (key.startsWith(SCREENING_KEY_PREFIX) && message) {
   328→        return { id: key.slice(SCREENING_KEY_PREFIX.length), message };
   329→      }
   330→    }
   331→    return null;
   332→  }

   334→  function goToStep(next: number) {
   335→    setDirection(next > step ? 1 : -1);
   336→    setStep(next);
   337→  }
   338→
   339→  function goNext() {
   340→    const next = step === 0 ? validateStep1() : validateStep2();
   341→    setErrors((prev) => ({ ...prev, ...next }));
   342→    if (Object.values(next).some(Boolean)) {
   343→      if (step === 1) {
   344→        // Screening wajib kosong: sorot + scroll + toast (pola error wizard).
   345→        const screeningError = firstScreeningErrorOf(next);
   346→        if (screeningError) {
   347→          scrollToScreening(screeningError.id);
   348→          toast.error(screeningError.message);
   349→        }
   350→      }
   351→      return;
   352→    }
   353→    goToStep(Math.min(2, step + 1));
   354→  }
   355→
   356→  function goBack() {
   357→    goToStep(Math.max(0, step - 1));
   358→  }

   360→  /** Validasi seluruh formulir; bila ada galat, lompat ke langkah pertama yang bermasalah. */
   361→  function validateAllAndJump(): boolean {
   362→    const e1 = validateStep1();
   363→    const e2 = validateStep2();
   364→    setErrors((prev) => ({ ...prev, ...e1, ...e2 }));
   365→    if (e1.name || e1.email || e1.phone || e1.positionId) {
   366→      goToStep(0);
   367→      return false;
   368→    }
   369→    if (Object.values({ ...e1, ...e2 }).some(Boolean)) {
   370→      goToStep(1);
   371→      // Konten langkah 1 baru muncul setelah transisi AnimatePresence selesai —
   372→      // scroll ke pertanyaan screening bermasalah dengan sedikit jeda.
   373→      const screeningError = firstScreeningErrorOf(e2);
   374→      if (screeningError) {
   375→        window.setTimeout(() => scrollToScreening(screeningError.id), 400);
   376→        toast.error(screeningError.message);
   377→      }
   378→      return false;
   379→    }
   380→    return true;
   381→  }

   398→  function acceptCv(file: File | null) {
   399→    if (!file) return;
   400→    if (file.type !== "application/pdf") {
   401→      setCvError(t.apply.errors.cvType);
   402→      toast.error(t.apply.errors.cvType);
   403→      return;
   404→    }
   405→    if (file.size > CV_MAX_BYTES) {
   406→      setCvError(t.apply.errors.cvSize);
   407→      toast.error(t.apply.errors.cvSize);
   408→      return;
   409→    }
   410→    setCvFile(file);
   411→    setCvError(null);
   412→  }

   414→  function acceptIntro(file: File | null) {
   415→    if (!file) return;
   416→    const typeOk =
   417→      file.type.startsWith("audio/") || /\.(mp3|wav|m4a)$/i.test(file.name);
   418→    if (!typeOk) {
   419→      setIntroError(t.apply.errors.introType);
   420→      toast.error(t.apply.errors.introType);
   421→      return;

   423→    if (file.size > INTRO_MAX_BYTES) {
   424→      setIntroError(t.apply.errors.introSize);
   425→      toast.error(t.apply.errors.introSize);
   426→      return;
   427→    }
   428→    setIntroFile(file);
   429→    setIntroError(null);
   430→  }

   457→  async function doSubmit() {
   458→    setSubmitting(true);
   459→    try {
   460→      const fd = new FormData();
   461→      fd.append("name", values.name.trim());
   462→      fd.append("email", values.email.trim());
   463→      fd.append("phone", values.phone.trim());
   464→      if (positionId) fd.append("positionId", positionId);
   465→      if (values.portfolioUrl.trim())
   466→        fd.append("portfolioUrl", values.portfolioUrl.trim());
   467→      if (values.socialLinks.trim())
   468→        fd.append("socialLinks", values.socialLinks.trim());
   469→      fd.append("experience", values.experience.trim());
   470→      fd.append("motivation", values.motivation.trim());
   471→      // Jawaban screening posisi (JSON {questionId: jawaban}) — hanya yang terisi.
   472→      if (screeningQuestions.length > 0) {
   473→        const record: Record<string, string> = {};
   474→        for (const question of screeningQuestions) {
   475→          const answer = (screeningAnswers[question.id] ?? "").trim();
   476→          if (answer) record[question.id] = answer.slice(0, SCREENING_MAX);
   477→        }
   478→        fd.append("screeningAnswers", JSON.stringify(record));
   479→      }
   480→      // Sumber pelamar + UTM dari URL saat halaman dibuka.
   481→      if (source) fd.append("source", source);
   482→      if (utm.source) fd.append("utmSource", utm.source);
   483→      if (utm.medium) fd.append("utmMedium", utm.medium);
   484→      if (utm.campaign) fd.append("utmCampaign", utm.campaign);
   485→      if (cvFile) fd.append("cvFile", cvFile);
   486→      if (introFile) fd.append("introFile", introFile);
   487→
   488→      const res = await fetch("/api/applications", { method: "POST", body: fd });
   489→      const data: unknown = await res.json().catch(() => null);
   490→      if (!res.ok) {
   491→        // Termasuk 409 kuota penuh — pesan spesifik dari server ditampilkan apa adanya.
   492→        const serverError =
   493→          typeof data === "object" &&
   494→          data !== null &&
   495→          "error" in data &&
   496→          typeof (data as { error: unknown }).error === "string"
   497→            ? (data as { error: string }).error
   498→            : null;
   499→        toast.error(serverError ?? t.apply.errors.submitFailed);
   500→        return;
   501→      }
   502→      const successData: ApplySuccessResponse | null =
   503→        typeof data === "object" &&
   504→        data !== null &&
   505→        "ok" in data &&
   506→        (data as { ok: unknown }).ok === true
   507→          ? (data as ApplySuccessResponse)
   508→          : null;
   509→      const trackingCode = successData?.trackingCode ?? "";
   510→
   511→      submittedRef.current = true;
   512→      try {
   513→        window.localStorage.removeItem(DRAFT_KEY);
   514→      } catch {
   515→        // abaikan
   516→      }
   517→      setSuccess({
   518→        name: values.name.trim(),
   519→        trackingCode,
   520→        autoReply: successData?.autoReply ?? null,
   521→        assignment: successData?.assignment ?? null,
   522→      });
   523→      toast.success(t.apply.success.title);
   524→    } catch {
   525→      toast.error(t.apply.errors.submitFailed);
   526→    } finally {
   527→      setSubmitting(false);
   528→    }
   529→  }

   531→  function handleSubmit(event: FormEvent<HTMLFormElement>) {
   532→    event.preventDefault();
   533→    if (submitting) return;
   534→
   535→    // Langkah 0-1: validasi per langkah lalu maju.
   536→    if (step === 0 || step === 1) {
   537→      goNext();
   538→      return;
   539→    }
   540→
   541→    // Langkah 3 (Berkas): validasi semuanya (termasuk berkas wajib), lalu pratinjau.
   542→    if (step === 2) {
   543→      if (!validateAllAndJump()) return;
   544→      if (!validateRequiredFiles()) return;
   545→      goToStep(3);
   546→      return;
   547→    }
   548→
   549→    // Langkah 3 (Pratinjau): wajib pernyataan kebenaran data sebelum konfirmasi.
   550→    if (!agreed) {
   551→      toast.error(t.apply.preview.agreeRequired);
   552→      return;
   553→    }
   554→    setConfirmOpen(true);
   555→  }

   557→  function resetForm() {
   558→    submittedRef.current = false;
   559→    draftDismissedRef.current = false;
   560→    setValues(INITIAL_VALUES);
   561→    setErrors({});
   562→    setStep(0);
   563→    setSuccess(null);
   564→    setCvFile(null);
   565→    setIntroFile(null);
   566→    setCvError(null);
   567→    setIntroError(null);
   568→    setAgreed(false);
   569→    setConfirmOpen(false);
   570→    setDirection(1);
   571→    setSource("");
   572→    setScreeningAnswers({});
