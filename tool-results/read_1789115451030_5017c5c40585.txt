    90→    90→function moveItem<T>(items: T[], index: number, direction: -1 | 1): T[] {
    91→    91→  const next = [...items];
    92→    92→  const target = index + direction;
    93→    93→  if (target < 0 || target >= next.length) return next;
    94→    94→  const [item] = next.splice(index, 1);
    95→    95→  next.splice(target, 0, item);
    96→    96→  return next;
    97→    97→}
