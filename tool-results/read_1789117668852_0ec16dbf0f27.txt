   803→/** JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions). */
   804→function useJobPostingJsonLd(positions: Position[], siteName: string) {
   805→  useEffect(() => {
   806→    const jobPostings = positions.map((position) => ({
   807→      "@context": "https://schema.org",
   808→      "@type": "JobPosting",
   809→      title: position.title,
   810→      description: position.description,
   811→      hiringOrganization: { "@type": "Organization", name: siteName },
   812→      jobLocationType: "TELECOMMUTE",
   813→      employmentType: employmentTypeOf(position.type),
   814→      datePosted: position.createdAt,
   815→      ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   816→    }));
   817→    const script = document.createElement("script");
   818→    script.id = "jobposting-ld";
   819→    script.type = "application/ld+json";
   820→    script.textContent = JSON.stringify(jobPostings);
   821→    document.head.appendChild(script);
   822→    return () => {
   823→      document.getElementById("jobposting-ld")?.remove();
   824→    };
   825→  }, [positions, siteName]);
   826→}

   828→function LandingShell({ content, positions, stats }: LandingPageProps) {

   837→    document.getElementById("lamar")?.scrollIntoView({ behavior: "smooth" });
   838→  };
   839→
   840→  return (
   841→    <div className="flex min-h-screen flex-col">

   844→        tagline={content.tagline}
   845→        sections={sections}

   849→          <Hero content={content} stats={stats} sections={sections} />
