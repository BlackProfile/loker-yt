     1→"use client";

     3→import { useCallback, useEffect, useMemo, useState } from "react";
     4→import { Button, buttonVariants } from "@/components/ui/button";
     5→import { Card, CardContent } from "@/components/ui/card";
     6→import {
     7→  AlertDialog,

     9→  AlertDialogCancel,
    10→  AlertDialogContent,

    18→  Select,
    19→  SelectContent,
    20→  SelectItem,
    21→  SelectTrigger,
    22→  SelectValue,
    23→} from "@/components/ui/select";
    24→import { Skeleton } from "@/components/ui/skeleton";
    25→import { Switch } from "@/components/ui/switch";
    26→import {
    27→  Download,
    28→  Inbox,
    29→  LayoutGrid,
    30→  Loader2,
    31→  RotateCcw,

    33→  Table2,
    34→  Trash2,

    37→import {
    38→  APPLICATION_SOURCES,
    39→  type Application,
    40→  type Position,
    41→  type StageKey,
    42→} from "@/lib/types";
    43→import {
    44→  DEFAULT_STAGES,
    45→  OTHER_STAGE_KEY,
    46→  isBuiltInStage,
    47→  stageLabel,
    48→  stagesForPosition,
    49→} from "@/lib/stages";
    50→import { apiDelete, apiGet, apiPatch, apiPost, buildQuery } from "./api";
    51→import { useAdminSession } from "./admin-context";
    52→import { ApplicationDetailDialog } from "./application-detail-dialog";

    54→import { KanbanBoard } from "./kanban-board";

    58→
    59→const ALL = "ALL";
    60→
    61→const FILTER_TRIGGER_CLASS = "h-10 w-full rounded-xl";
    62→
    63→const SORT_OPTIONS = [
    64→  { value: "newest", label: "Terbaru" },
    65→  { value: "oldest", label: "Terlama" },
    66→  { value: "aiScore", label: "Skor AI" },
    67→] as const;
    68→
    69→type ViewMode = "table" | "kanban";
    70→
    71→export function ApplicationsTab() {
    72→  const { canMutate, reportError } = useAdminSession();
    73→
    74→  const [applications, setApplications] = useState<Application[]>([]);
    75→  const [positions, setPositions] = useState<Position[]>([]);
    76→  const [loading, setLoading] = useState(true);
    77→
    78→  const [searchInput, setSearchInput] = useState("");
    79→  const [q, setQ] = useState("");
    80→  const [statusFilter, setStatusFilter] = useState<string>(ALL);
    81→  const [positionFilter, setPositionFilter] = useState<string>(ALL);
    82→  const [sourceFilter, setSourceFilter] = useState<string>(ALL);
    83→  const [sort, setSort] = useState<string>("newest");
    84→  const [ratingMin, setRatingMin] = useState<string>("");
    85→  const [tag, setTag] = useState<string>(ALL);
    86→  const [talentPool, setTalentPool] = useState(false);
    87→  const [hasInterview, setHasInterview] = useState(false);
    88→
    89→  const [allTags, setAllTags] = useState<string[]>([]);
    90→  const [view, setView] = useState<ViewMode>("table");
    91→
    92→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    93→  const [compareIds, setCompareIds] = useState<string[]>([]);
    94→  const [bulkStatus, setBulkStatus] = useState<string>("");
    95→  const [bulkWorking, setBulkWorking] = useState(false);
    96→  const [bulkDeleteOpen, setBulkDeleteOpen] = useState(false);
    97→
    98→  const [detail, setDetail] = useState<Application | null>(null);
    99→  const [deleteTarget, setDeleteTarget] = useState<Application | null>(null);
   100→  const [deleting, setDeleting] = useState(false);
   101→  const [compareOpen, setCompareOpen] = useState(false);
   102→
   103→  // Posisi terpilih menentukan opsi tahap (pipeline kustom vs bawaan).
   104→  const selectedPosition = useMemo(
   105→    () =>
   106→      positionFilter !== ALL
   107→        ? positions.find((p) => p.id === positionFilter) ?? null
   108→        : null,
   109→    [positions, positionFilter]
   110→  );
   111→
   112→  // Opsi filter tahap: dengan posisi -> tahap milik posisi;
   113→  // tanpa posisi -> 5 bawaan + "Lainnya" (tahap kustom, disaring client-side).
   114→  const stageOptions = useMemo<StageKey[]>(
   115→    () =>
   116→      selectedPosition
   117→        ? stagesForPosition(selectedPosition.stages)
   118→        : [...DEFAULT_STAGES, OTHER_STAGE_KEY],
   119→    [selectedPosition]
   120→  );
   121→
   122→  // Tahap nyata untuk bulk change (tanpa opsi "Lainnya").
   123→  const bulkStageOptions = useMemo<StageKey[]>(
   124→    () =>
   125→      selectedPosition ? stagesForPosition(selectedPosition.stages) : [...DEFAULT_STAGES],
   126→    [selectedPosition]
   127→  );
   128→
   129→  const isOtherStageFilter = statusFilter === OTHER_STAGE_KEY;
   130→
   131→  const activeQuery = useMemo(() => {
   132→    return buildQuery({
   133→      q: q || undefined,
   134→      // "Lainnya" tidak dikirim ke server: fetch tanpa filter status,
   135→      // lalu disaring client-side (status bukan 5 bawaan).
   136→      status:
   137→        statusFilter !== ALL && !isOtherStageFilter ? statusFilter : undefined,
   138→      positionId: positionFilter !== ALL ? positionFilter : undefined,
   139→      source: sourceFilter !== ALL ? sourceFilter : undefined,
   140→      sort: sort !== "newest" ? sort : undefined,
   141→      ratingMin: ratingMin !== "" ? ratingMin : undefined,
   142→      tag: tag !== ALL ? tag : undefined,
   143→      talentPool: talentPool ? "1" : undefined,
   144→      hasInterview: hasInterview ? "1" : undefined,
   145→    });
   146→  }, [
   147→    q,
   148→    statusFilter,
   149→    isOtherStageFilter,
   150→    positionFilter,
   151→    sourceFilter,
   152→    sort,
   153→    ratingMin,
   154→    tag,
   155→    talentPool,
   156→    hasInterview,
   157→  ]);
   158→
   159→  const hasActiveFilter =
   160→    q !== "" ||
   161→    statusFilter !== ALL ||
   162→    positionFilter !== ALL ||
   163→    sourceFilter !== ALL ||
   164→    sort !== "newest" ||
   165→    ratingMin !== "" ||
   166→    tag !== ALL ||
   167→    talentPool ||
   168→    hasInterview;
   169→
   170→  const loadPositions = useCallback(async () => {
   171→    try {
   172→      const data = await apiGet<Position[]>("/api/admin/positions");
   173→      setPositions(data);
   174→    } catch {
   175→      // Filter posisi opsional; abaikan error fetch daftar posisi.
   176→    }
   177→  }, []);
   178→
   179→  const loadApplications = useCallback(async () => {
   180→    setLoading(true);
   181→    try {
   182→      const data = await apiGet<Application[]>(
   183→        `/api/admin/applications${activeQuery}`
   184→      );
   185→      setApplications(data);
   186→      // Kumpulkan tag unik untuk pilihan filter.
   187→      setAllTags((prev) => {
   188→        const set = new Set(prev);
   189→        for (const app of data) for (const t of app.tags) set.add(t);
   190→        return Array.from(set).sort((a, b) => a.localeCompare(b, "id"));
   191→      });
   192→    } catch (err) {
   193→      reportError(err);
   194→    } finally {
   195→      setLoading(false);
   196→    }
   197→  }, [activeQuery, reportError]);
   198→
   199→  useEffect(() => {
   200→    void loadPositions();
   201→  }, [loadPositions]);
   202→
   203→  useEffect(() => {
   204→    void loadApplications();
   205→  }, [loadApplications]);
   206→
   207→  function applySearch() {
   208→    setQ(searchInput.trim());
   209→  }
   210→
   211→  // Saat konteks posisi berubah, buang filter tahap yang tak berlaku lagi
   212→  // (mis. tahap kustom posisi lain atau opsi "Lainnya").
   213→  function changePositionFilter(value: string) {
   214→    setPositionFilter(value);
   215→    if (value === ALL) {
   216→      // Kembali ke semua posisi: hanya tahap bawaan yang tetap berlaku.
   217→      if (statusFilter !== ALL && !isBuiltInStage(statusFilter)) {
   218→        setStatusFilter(ALL);
   219→      }
   220→      return;
   221→    }
   222→    const position = positions.find((p) => p.id === value);
   223→    const options = position ? stagesForPosition(position.stages) : [];
   224→    if (statusFilter !== ALL && !options.includes(statusFilter)) {
   225→      setStatusFilter(ALL);
   226→    }
   227→  }
   228→
   229→  function resetFilters() {
   230→    setSearchInput("");
   231→    setQ("");
   232→    setStatusFilter(ALL);
   233→    setPositionFilter(ALL);
   234→    setSourceFilter(ALL);
   235→    setSort("newest");
   236→    setRatingMin("");
   237→    setTag(ALL);
   238→    setTalentPool(false);
   239→    setHasInterview(false);
   240→  }
   241→
   242→  function toggleSelect(id: string, checked: boolean) {
   243→    setSelectedIds((prev) => {
   244→      const next = new Set(prev);
   245→      if (checked) next.add(id);
   246→      else next.delete(id);
   247→      return next;
   248→    });
   249→  }
   250→
   251→  // Lamaran tampil: dengan filter "Lainnya", sisakan yang tahapnya di luar 5 bawaan.
   252→  const displayedApplications = useMemo(() => {
   253→    if (!isOtherStageFilter) return applications;
   254→    return applications.filter((a) => !isBuiltInStage(a.status));
   255→  }, [applications, isOtherStageFilter]);
   256→
   257→  function toggleSelectAll(checked: boolean) {
   258→    setSelectedIds((prev) => {
   259→      const next = new Set(prev);
   260→      if (checked) {
   261→        for (const app of displayedApplications) next.add(app.id);
   262→      } else {
   263→        for (const app of displayedApplications) next.delete(app.id);
   264→      }
   265→      return next;
   266→    });
   267→  }
   268→
   269→  function toggleCompare(app: Application) {
   270→    setCompareIds((prev) => {
   271→      if (prev.includes(app.id)) {
   272→        return prev.filter((id) => id !== app.id);
   273→      }
   274→      if (prev.length >= 3) {
   275→        toast.info("Maksimal 3 kandidat — kandidat pertama diganti.");
   276→        return [...prev.slice(1), app.id];
   277→      }
   278→      return [...prev, app.id];
   279→    });
   280→  }
   281→
   282→  function updateAppInList(updated: Application) {
   283→    setApplications((prev) => prev.map((a) => (a.id === updated.id ? updated : a)));
   284→    setDetail((prev) => (prev && prev.id === updated.id ? updated : prev));
   285→  }
   286→
   287→  async function handleRate(app: Application, rating: number) {
   288→    try {
   289→      const updated = await apiPatch<Application>(
   290→        `/api/admin/applications/${app.id}`,
   291→        { rating }
   292→      );
   293→      toast.success(`Rating disimpan (${rating}/5)`);
   294→      updateAppInList(updated);
   295→    } catch (err) {
   296→      reportError(err);
   297→    }
   298→  }
   299→
   300→  // Pindah kolom kanban: optimistik + PATCH, revert bila gagal.
   301→  function handleKanbanMove(id: string, status: StageKey) {
   302→    const previous = applications;
   303→    const target = stageLabel(status);
   304→    setApplications((prev) =>
   305→      prev.map((a) => (a.id === id ? { ...a, status } : a))
   306→    );
   307→    apiPatch<Application>(`/api/admin/applications/${id}`, { status })
   308→      .then((updated) => {
   309→        toast.success(`Status diubah ke ${target}`);
   310→        updateAppInList(updated);
   311→      })
   312→      .catch((err) => {
   313→        setApplications(previous);
   314→        reportError(err);
   315→      });
   316→  }
   317→
   318→  async function runBulk(body: Record<string, unknown>, successMessage: string) {
   319→    if (bulkWorking) return;
   320→    setBulkWorking(true);
   321→    try {
   322→      const res = await apiPost<{ ok: boolean; affected: number }>(
   323→        "/api/admin/applications/bulk",
   324→        body
   325→      );
   326→      toast.success(successMessage.replace("{n}", String(res.affected)));
   327→      setSelectedIds(new Set());
   328→      setBulkStatus("");
   329→      await loadApplications();
   330→    } catch (err) {
   331→      reportError(err);
   332→    } finally {
   333→      setBulkWorking(false);
   334→      setBulkDeleteOpen(false);
   335→    }
   336→  }
   337→
   338→  async function handleDelete() {
   339→    if (!deleteTarget || deleting) return;
   340→    setDeleting(true);
   341→    try {
   342→      await apiDelete<{ ok: boolean }>(
   343→        `/api/admin/applications/${deleteTarget.id}`
   344→      );
   345→      toast.success("Lamaran dihapus");
   346→      setApplications((prev) => prev.filter((a) => a.id !== deleteTarget.id));
   347→      setDetail(null);
   348→    } catch (err) {
   349→      reportError(err);
   350→    } finally {
   351→      setDeleting(false);
   352→      setDeleteTarget(null);
   353→    }
   354→  }
   355→
   356→  const comparedApps = useMemo(
   357→    () =>
   358→      compareIds
   359→        .map((id) => displayedApplications.find((a) => a.id === id))
   360→        .filter((a): a is Application => Boolean(a)),
   361→    [compareIds, displayedApplications]
   362→  );
   363→
   364→  return (
   365→    <div className="flex flex-col gap-4">
   366→      {/* Toolbar filter */}
   367→      <div className="flex flex-col gap-3">
   368→        <div className="flex flex-col gap-2 lg:flex-row lg:items-center">
   369→          <div className="relative flex-1">
   370→            <Search
   371→              className="text-muted-foreground pointer-events-none absolute top-1/2 left-3 size-4 -translate-y-1/2"
   372→              aria-hidden="true"
   373→            />
   374→            <Input
   375→              value={searchInput}
   376→              onChange={(e) => setSearchInput(e.target.value)}
   377→              onKeyDown={(e) => {
   378→                if (e.key === "Enter") applySearch();
   379→              }}
   380→              onBlur={applySearch}
   381→              placeholder="Cari nama atau email..."
   382→              aria-label="Cari nama atau email pelamar"
   383→              className="h-10 rounded-xl pl-9"
   384→            />
   385→          </div>
   386→          <div className="flex flex-wrap items-center gap-2">
   387→            {/* Toggle tampilan */}
   388→            <div className="flex items-center gap-1 rounded-xl border p-1">
   389→              <Button
   390→                variant={view === "table" ? "secondary" : "ghost"}
   391→                size="sm"
   392→                className={cn("h-8", view === "table" && "shadow-xs")}
   393→                onClick={() => setView("table")}
   394→                aria-pressed={view === "table"}
   395→              >
   396→                <Table2 className="size-4" aria-hidden="true" />
   397→                Tabel
   398→              </Button>
   399→              <Button
   400→                variant={view === "kanban" ? "secondary" : "ghost"}
   401→                size="sm"
   402→                className={cn("h-8", view === "kanban" && "shadow-xs")}
   403→                onClick={() => setView("kanban")}
   404→                aria-pressed={view === "kanban"}
   405→              >
   406→                <LayoutGrid className="size-4" aria-hidden="true" />
   407→                Kanban
   408→              </Button>
   409→            </div>
   410→            {/* Export CSV */}
   411→            <a
   412→              href={`/api/admin/applications/export${activeQuery}`}
   413→              download
   414→              className={cn(buttonVariants({ variant: "outline" }), "h-10 rounded-xl")}
   415→              aria-label="Ekspor daftar lamaran ke CSV"
   416→            >
   417→              <Download className="size-4" aria-hidden="true" />
   418→              <span className="hidden sm:inline">Export CSV</span>
   419→            </a>
   420→          </div>
   421→        </div>
   422→
   423→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-3 lg:grid-cols-6">
   424→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   425→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Filter tahap">
   426→              <SelectValue placeholder="Semua Tahap" />
   427→            </SelectTrigger>
   428→            <SelectContent>
   429→              <SelectItem value={ALL}>Semua Tahap</SelectItem>
   430→              {stageOptions.map((s) => (
   431→                <SelectItem key={s} value={s}>
   432→                  {s === OTHER_STAGE_KEY ? "Lainnya (tahap kustom)" : stageLabel(s)}
   433→                </SelectItem>
   434→              ))}
   435→            </SelectContent>
   436→          </Select>
   437→
   438→          <Select
   439→            value={positionFilter}
   440→            onValueChange={changePositionFilter}
   441→          >
   442→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Filter posisi">
   443→              <SelectValue placeholder="Semua Posisi" />
   444→            </SelectTrigger>
   445→            <SelectContent>
   446→              <SelectItem value={ALL}>Semua Posisi</SelectItem>
   447→              {positions.map((p) => (
   448→                <SelectItem key={p.id} value={p.id}>
   449→                  {p.title}
   450→                </SelectItem>
   451→              ))}
   452→            </SelectContent>
   453→          </Select>
   454→
   455→          <Select value={sourceFilter} onValueChange={setSourceFilter}>
   456→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Filter sumber pelamar">
   457→              <SelectValue placeholder="Semua Sumber" />
   458→            </SelectTrigger>
   459→            <SelectContent>
   460→              <SelectItem value={ALL}>Semua Sumber</SelectItem>
   461→              {APPLICATION_SOURCES.map((s) => (
   462→                <SelectItem key={s} value={s}>
   463→                  {s}
   464→                </SelectItem>
   465→              ))}
   466→            </SelectContent>
   467→          </Select>
   468→
   469→          <Select value={sort} onValueChange={setSort}>
   470→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Urutkan">
   471→              <SelectValue placeholder="Urutkan" />
   472→            </SelectTrigger>
   473→            <SelectContent>
   474→              {SORT_OPTIONS.map((opt) => (
   475→                <SelectItem key={opt.value} value={opt.value}>
   476→                  {opt.label}
   477→                </SelectItem>
   478→              ))}
   479→            </SelectContent>
   480→          </Select>
   481→
   482→          <Select value={ratingMin} onValueChange={setRatingMin}>
   483→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Filter rating minimum">
   484→              <SelectValue placeholder="Semua Rating" />
   485→            </SelectTrigger>
   486→            <SelectContent>
   487→              <SelectItem value="all-rating">Semua Rating</SelectItem>
   488→              {[1, 2, 3, 4, 5].map((n) => (
   489→                <SelectItem key={n} value={String(n)}>
   490→                  Rating &#8805; {n}
   491→                </SelectItem>
   492→              ))}
   493→            </SelectContent>
   494→          </Select>
   495→
   496→          <Select
   497→            value={tag === ALL ? "all-tag" : tag}
   498→            onValueChange={(v) => setTag(v === "all-tag" ? ALL : v)}
   499→          >
   500→            <SelectTrigger className={FILTER_TRIGGER_CLASS} aria-label="Filter tag">
   501→              <SelectValue placeholder="Semua Tag" />
   502→            </SelectTrigger>
   503→            <SelectContent>
   504→              <SelectItem value="all-tag">Semua Tag</SelectItem>
   505→              {allTags.map((t) => (
   506→                <SelectItem key={t} value={t}>
   507→                  {t}
   508→                </SelectItem>
   509→              ))}
   510→            </SelectContent>
   511→          </Select>
   512→        </div>
   513→
   514→        <div className="flex flex-wrap items-center gap-4">
   515→          <label className="flex cursor-pointer items-center gap-2 text-xs font-medium">
   516→            <Switch
   517→              checked={talentPool}
   518→              onCheckedChange={setTalentPool}
   519→              aria-label="Filter Talent Pool"
   520→            />
   521→            Talent Pool
   522→          </label>
   523→          <label className="flex cursor-pointer items-center gap-2 text-xs font-medium">
   524→            <Switch
   525→              checked={hasInterview}
   526→              onCheckedChange={setHasInterview}
   527→              aria-label="Filter ada jadwal wawancara"
   528→            />
   529→            Ada Jadwal Wawancara
   530→          </label>
   531→          {hasActiveFilter ? (
   532→            <Button
   533→              variant="outline"
   534→              onClick={resetFilters}
   535→              className="h-9 rounded-xl"
   536→              aria-label="Reset filter"
   537→            >
   538→              <RotateCcw className="size-4" aria-hidden="true" />
   539→              Reset Filter
   540→            </Button>
   541→          ) : null}
   542→          <p className="ml-auto text-xs text-muted-foreground">
   543→            {loading
   544→              ? "Memuat..."
   545→              : `${displayedApplications.length} lamaran ditampilkan`}
   546→          </p>
   547→        </div>
   548→      </div>
   549→
   550→      {/* Loading */}
   551→      {loading ? (
   552→        <div className="flex flex-col gap-3">
   553→          {Array.from({ length: 5 }).map((_, i) => (
   554→            <Skeleton key={i} className="h-14 w-full rounded-xl" />
   555→          ))}
   556→        </div>
   557→      ) : applications.length === 0 ? (
   558→        <Card className="rounded-2xl">
   559→          <CardContent className="flex flex-col items-center gap-2 py-14 text-center">
   560→            <Inbox className="size-10 text-muted-foreground/50" aria-hidden="true" />
   561→            <p className="text-sm text-muted-foreground">
   562→              Belum ada lamaran yang cocok dengan filter.
   563→            </p>
   564→          </CardContent>
   565→        </Card>
   566→      ) : view === "table" ? (
   567→        <ApplicationsTable
   568→          applications={displayedApplications}
   569→          canMutate={canMutate}
   570→          selectedIds={selectedIds}
   571→          onToggleSelect={toggleSelect}
   572→          onToggleSelectAll={toggleSelectAll}
   573→          compareIds={compareIds}
   574→          onToggleCompare={toggleCompare}
   575→          onOpenDetail={setDetail}
   576→          onDeleteRequest={setDeleteTarget}
   577→          onRate={(app, rating) => void handleRate(app, rating)}
   578→        />
   579→      ) : (
   580→        <KanbanBoard
   581→          apps={displayedApplications}
   582→          canMutate={canMutate}
   583→          stages={selectedPosition?.stages}
   584→          hasPositionFilter={positionFilter !== ALL}
   585→          onMove={handleKanbanMove}
   586→          onOpenDetail={setDetail}
   587→        />
   588→      )}
   589→
   590→      {/* Bulk bar */}
   591→      {canMutate && selectedIds.size > 0 ? (
   592→        <Reveal
   593→          slideY={8}
   594→          duration={0.2}
   595→          className="sticky bottom-4 z-20"
   596→        >
   597→          <div className="flex flex-wrap items-center gap-2 rounded-2xl border border-rose-200 bg-rose-50/95 p-3 backdrop-blur dark:border-rose-900 dark:bg-rose-950/90">
   598→          <span className="text-sm font-semibold">
   599→            {selectedIds.size} dipilih

   605→            >
   606→              <SelectValue placeholder="Ubah tahap ke..." />
   607→            </SelectTrigger>

   609→              <SelectItem value="bulk-empty" disabled>
   610→                Ubah tahap ke...
   611→              </SelectItem>

   614→                  {stageLabel(s)}

   617→            </SelectContent>
   618→          </Select>
   619→          <Button
   620→            size="sm"
   621→            className="h-9 active:scale-[0.99]"
   622→            disabled={!bulkStatus || bulkWorking}
   623→            onClick={() =>
   624→              void runBulk(
   625→                { ids: Array.from(selectedIds), action: "status", status: bulkStatus },
   626→                "{n} lamaran diperbarui"
   627→              )
   628→            }
   629→          >
   630→            {bulkWorking ? (
   631→              <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   632→            ) : null}
   633→            Terapkan
   634→          </Button>
   635→          <Button
   636→            variant="outline"
   637→            size="sm"
   638→            className="h-9 bg-background"
   639→            disabled={bulkWorking}
   640→            onClick={() =>
   641→              void runBulk(
   642→                { ids: Array.from(selectedIds), action: "talentPool", talentPool: true },
   643→                "{n} lamaran masuk Talent Pool"
   644→              )
   645→            }
   646→          >
   647→            Talent Pool
   648→          </Button>
   649→          <Button
   650→            variant="destructive"
   651→            size="sm"
   652→            className="h-9"
   653→            disabled={bulkWorking}
   654→            onClick={() => setBulkDeleteOpen(true)}
   655→          >
   656→            <Trash2 className="size-4" aria-hidden="true" />

   658→          </Button>

   660→            variant="ghost"
   661→            size="sm"
   662→            className="h-9"
   663→            onClick={() => setSelectedIds(new Set())}
   664→            disabled={bulkWorking}
   665→          >
   666→            Bersihkan pilihan
   667→          </Button>
   668→          </div>
   669→        </Reveal>
   670→      ) : null}
   671→

   677→            duration={0.25}

   679→            whileTap={{ scale: 0.97 }}
   680→            className="pointer-events-auto"
   681→          >
   682→            <Button
   683→              className="h-11 rounded-full px-5 shadow-lg"
   684→              onClick={() => setCompareOpen(true)}

   687→              Bandingkan ({comparedApps.length})
   688→            </Button>
   689→          </Reveal>
   690→        </div>
   691→      ) : null}
   692→
   693→      <ComparisonDialog
   694→        apps={compareOpen ? comparedApps : []}
   695→        onOpenChange={setCompareOpen}
   696→      />
   697→
   698→      <ApplicationDetailDialog
   699→        application={detail}

   703→        onSaved={updateAppInList}
   704→        onDeleted={(id) =>
   705→          setApplications((prev) => prev.filter((a) => a.id !== id))
   706→        }
   707→      />
   708→
   709→      {/* Konfirmasi hapus dari baris */}
   710→      <AlertDialog
   711→        open={!!deleteTarget}
   712→        onOpenChange={(open) => {
   713→          if (!open) setDeleteTarget(null);
   714→        }}
   715→      >

   721→                ? `Lamaran ${deleteTarget.name} akan dihapus permanen. Tindakan tidak bisa dibatalkan.`
   722→                : "Tindakan tidak bisa dibatalkan."}
   723→            </AlertDialogDescription>
   724→          </AlertDialogHeader>
   725→          <AlertDialogFooter>
   726→            <AlertDialogCancel disabled={deleting}>Batal</AlertDialogCancel>
   727→            <AlertDialogAction
   728→              onClick={(e) => {
   729→                e.preventDefault();
   730→                void handleDelete();
   731→              }}
   732→              className="bg-rose-600 text-white hover:bg-rose-700"

   740→
   741→      {/* Konfirmasi hapus massal */}
   742→      <AlertDialog open={bulkDeleteOpen} onOpenChange={setBulkDeleteOpen}>
   743→        <AlertDialogContent>
   744→          <AlertDialogHeader>
   745→            <AlertDialogTitle>Hapus {selectedIds.size} lamaran?</AlertDialogTitle>
   746→            <AlertDialogDescription>
   747→              Semua lamaran terpilih akan dihapus permanen. Tindakan tidak bisa
   748→              dibatalkan.
   749→            </AlertDialogDescription>
   750→          </AlertDialogHeader>
   751→          <AlertDialogFooter>
   752→            <AlertDialogCancel disabled={bulkWorking}>Batal</AlertDialogCancel>
   753→            <AlertDialogAction
   754→              onClick={(e) => {
   755→                e.preventDefault();
   756→                void runBulk(
   757→                  { ids: Array.from(selectedIds), action: "delete" },
   758→                  "{n} lamaran dihapus"
   759→                );
   760→              }}
   761→              className="bg-rose-600 text-white hover:bg-rose-700"
   762→              disabled={bulkWorking}
   763→            >
   764→              {bulkWorking ? "Menghapus..." : "Ya, Hapus Semua"}
   765→            </AlertDialogAction>
   766→          </AlertDialogFooter>
   767→        </AlertDialogContent>
   768→      </AlertDialog>
   769→
   770→    </div>
   771→  );
