   778→/** JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions). */
   779→function useJobPostingJsonLd(positions: Position[], siteName: string) {
   780→  useEffect(() => {
   781→    const jobPostings = positions.map((position) => ({
   782→      "@context": "https://schema.org",
   783→      "@type": "JobPosting",
   784→      title: position.title,
   785→      description: position.description,
   786→      hiringOrganization: { "@type": "Organization", name: siteName },
   787→      jobLocationType: "TELECOMMUTE",
   788→      employmentType: employmentTypeOf(position.type),
   789→      datePosted: position.createdAt,
   790→      ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   791→    }));
   792→    const script = document.createElement("script");
   793→    script.id = "jobposting-ld";
   794→    script.type = "application/ld+json";
   795→    script.textContent = JSON.stringify(jobPostings);
   796→    document.head.appendChild(script);
   797→    return () => {
   798→      document.getElementById("jobposting-ld")?.remove();
   799→    };
   800→  }, [positions, siteName]);
   801→}
