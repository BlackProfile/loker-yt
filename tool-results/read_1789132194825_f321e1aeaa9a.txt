    94→// Status "halaman sudah digulir" via useSyncExternalStore (murah, tanpa setState di effect).
    95→function useScrolled(threshold = 8): boolean {
    96→  return useSyncExternalStore(
    97→    (notify) => {
    98→      window.addEventListener("scroll", notify, { passive: true });
    99→      return () => window.removeEventListener("scroll", notify);
   100→    },
   101→    () => window.scrollY > threshold,
   102→    () => false,
   103→  );
   104→}

   872→/**
   873→ * JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions).
   874→ * Bila focusSlug cocok dengan sebuah posisi (deep link /?posisi=slug), emit SATU
   875→ * schema JobPosting posisi tersebut — lengkap dengan url deep link — else semua posisi.
   876→ */
   877→function useJobPostingJsonLd(
   878→  positions: Position[],
   879→  siteName: string,
   880→  focusSlug?: string | null,
   881→) {
   882→  useEffect(() => {
   883→    const focused = focusSlug
   884→      ? (positions.find((position) => position.slug === focusSlug) ?? null)
   885→      : null;
   886→    const jobPostings = focused
   887→      ? [
   888→          {
   889→            "@context": "https://schema.org",
   890→            "@type": "JobPosting",
   891→            title: focused.title,
   892→            description: focused.description.trim().slice(0, 300),
   893→            hiringOrganization: { "@type": "Organization", name: siteName },
   894→            jobLocationType: "TELECOMMUTE",
   895→            employmentType: employmentTypeOf(focused.type),
   896→            datePosted: focused.createdAt,
   897→            ...(focused.closesAt ? { validThrough: focused.closesAt } : {}),
   898→            ...(focused.slug ? { url: buildPositionUrl(focused) } : {}),
   899→          },
   900→        ]
   901→      : positions.map((position) => ({
   902→          "@context": "https://schema.org",
   903→          "@type": "JobPosting",
   904→          title: position.title,
   905→          description: position.description,
   906→          hiringOrganization: { "@type": "Organization", name: siteName },
   907→          jobLocationType: "TELECOMMUTE",
   908→          employmentType: employmentTypeOf(position.type),
   909→          datePosted: position.createdAt,
   910→          ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   911→        }));
   912→    const script = document.createElement("script");
   913→    script.id = "jobposting-ld";
   914→    script.type = "application/ld+json";
   915→    script.textContent = JSON.stringify(jobPostings);
   916→    document.head.appendChild(script);
   917→    return () => {
   918→      document.getElementById("jobposting-ld")?.remove();
   919→    };
   920→  }, [positions, siteName, focusSlug]);
   921→}

   923→function LandingShell({
   924→  content,
   925→  positions,
   926→  stats,
   927→  positionStats,
   928→  initialPosisiSlug,
   929→}: LandingPageProps) {

   931→  // Visibilitas tiap bagian halaman publik (dikendalikan dari panel admin).

   935→

   937→    setSelectedPositionId(positionId);
   938→    document.getElementById("lamar")?.scrollIntoView({ behavior: "smooth" });
