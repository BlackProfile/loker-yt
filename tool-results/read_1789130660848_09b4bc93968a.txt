   211→function formPublicationMode(form: FormState): {
   212→  key: keyof typeof PUB_MODE;
   213→  hint: string;
   214→} {
   215→  if (!form.isActive) {
   216→    return { key: "draft", hint: "Posisi tidak tampil di halaman publik." };
   217→  }
   218→  if (form.publishAtLocal) {
   219→    const t = new Date(form.publishAtLocal).getTime();
   220→    if (!Number.isNaN(t) && t > Date.now()) {
   221→      return {
   222→        key: "terjadwal",
   223→        hint: "Posisi otomatis tayang tepat pada jadwal publikasi di atas.",
   224→      };
   225→    }
   226→  }
   227→  if (form.closesAtLocal) {
   228→    const t = new Date(form.closesAtLocal).getTime();
   229→    if (!Number.isNaN(t) && t <= Date.now()) {
   230→      return {
   231→        key: "tutup",
   232→        hint: "Posisi sudah melewati tanggal penutupan dan tidak menerima lamaran baru.",
   233→      };
   234→    }
   235→  }
   236→  return { key: "tayang", hint: "Posisi tampil di halaman publik sekarang." };
   237→}

   258→export function PositionFormDialog({
   259→  open,
   260→  onOpenChange,
   261→  editing,
   262→  statsRow,
   263→  onSaved,
   264→}: {
   265→  open: boolean;
   266→  onOpenChange: (open: boolean) => void;
   267→  editing: Position | null;
   268→  statsRow: PositionStatsRow | null;
   269→  onSaved: () => void;
   270→}) {
   271→  const { reportError } = useAdminSession();
   272→  const [form, setForm] = useState<FormState>(() =>
   273→    editing ? buildFormState(editing) : EMPTY_FORM
   274→  );
   275→  const [validationErrors, setValidationErrors] = useState<string[]>([]);
   276→  const [saving, setSaving] = useState(false);
   277→  const [uploadingCover, setUploadingCover] = useState(false);
   278→  const [aiCoverLoading, setAiCoverLoading] = useState(false);
   279→  const fileInputRef = useRef<HTMLInputElement>(null);
   280→
   281→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) =>
   282→    setForm((f) => ({ ...f, [key]: value }));
   283→
   284→  const cleanedStages = useMemo(
   285→    () => form.stages.map((s) => s.trim()).filter(Boolean),
   286→    [form.stages]
   287→  );
   288→
   289→  // Lamaran pada tahap di luar daftar pipeline tersimpan (dari stats funnel).
   290→  const outOfStageApps = useMemo(() => {
   291→    if (!editing || !statsRow) return 0;
   292→    const inFunnel = statsRow.funnel.reduce((sum, f) => sum + f.count, 0);
   293→    return Math.max(0, statsRow.applications - inFunnel);
   294→  }, [editing, statsRow]);
   295→
   296→  const pubMode = formPublicationMode(form);
   297→  const pub = PUB_MODE[pubMode.key];
   298→
   299→  function validate(): string[] {
   300→    const errors: string[] = [];
   301→    if (form.title.trim().length < 3)
   302→      errors.push("Judul posisi minimal 3 karakter.");
   303→    if (!form.department.trim()) errors.push("Departemen wajib diisi.");
   304→    if (form.description.trim().length < 10)
   305→      errors.push("Deskripsi minimal 10 karakter.");
   306→    if (form.salaryText.trim().length > 80)
   307→      errors.push("Teks gaji maksimal 80 karakter.");
   308→    if (
   309→      form.examples.some(
   310→        (url) => url.trim().length > 0 && !/^https?:\/\//i.test(url.trim())
   311→      )
   312→    )
   313→      errors.push("Contoh karya harus diawali http:// atau https://.");
   314→    if (
   315→      form.assignmentUrl.trim().length > 0 &&
   316→      !/^https?:\/\//i.test(form.assignmentUrl.trim())
   317→    )
   318→      errors.push("URL tes/assignment harus diawali http:// atau https://.");
   319→    if (form.screeningQuestions.some((q) => q.label.trim().length < 3))
   320→      errors.push("Label pertanyaan screening minimal 3 karakter.");
   321→    if (cleanedStages.length > 12)
   322→      errors.push("Pipeline tahap maksimal 12 tahap.");
   323→    if (form.maxApplicants.trim() !== "") {
   324→      if (!isInt(form.maxApplicants) || Number(form.maxApplicants) < 1 || Number(form.maxApplicants) > 10000)
   325→        errors.push("Kuota pelamar harus angka bulat 1-10000, atau dikosongkan.");
   326→    }
   327→    if (form.autoShortlistScore.trim() !== "") {
   328→      if (!isInt(form.autoShortlistScore) || Number(form.autoShortlistScore) < 0 || Number(form.autoShortlistScore) > 100)
   329→        errors.push("Skor auto-shortlist harus angka bulat 0-100, atau dikosongkan.");
   330→    }
   331→    if (form.order.trim() !== "" && !isInt(form.order))
   332→      errors.push("Urutan harus berupa bilangan bulat.");
   333→    return errors;
   334→  }
   335→
   336→  async function handleSave() {
   337→    if (saving) return;
   338→    const errors = validate();
   339→    setValidationErrors(errors);
   340→    if (errors.length > 0) {
   341→      toast.error(errors[0]);
   342→      return;
   343→    }
   344→    setSaving(true);
   345→    const payload: Record<string, unknown> = {
   346→      title: form.title.trim(),
   347→      department: form.department.trim(),
   348→      type: form.type,
   349→      location: form.location.trim() || "Remote",
   350→      description: form.description.trim(),
   351→      requirements: form.requirements.map((r) => r.trim()).filter(Boolean),
   352→      isActive: form.isActive,
   353→      publishAt: localInputToIso(form.publishAtLocal),
   354→      closesAt: localInputToIso(form.closesAtLocal),
   355→      coverFileId: form.coverFileId,
   356→      salaryText: form.salaryText.trim() || null,
   357→      salaryVisible: form.salaryVisible,
   358→      benefits: form.benefits.map((b) => b.trim()).filter(Boolean),
   359→      examples: form.examples.map((e) => e.trim()).filter(Boolean),
   360→      urgent: form.urgent,
   361→      featured: form.featured,
   362→      requireCv: form.requireCv,
   363→      requireIntro: form.requireIntro,
   364→      requirePortfolio: form.requirePortfolio,
   365→      maxApplicants:
   366→        form.maxApplicants.trim() === "" ? null : Number(form.maxApplicants),
   367→      screeningQuestions: form.screeningQuestions.map((q) => ({
   368→        id: q.id,
   369→        label: q.label.trim(),
   370→        required: q.required,
   371→      })),
   372→      stages: cleanedStages,
   373→      aiCriteria: form.aiCriteria.trim() || null,
   374→      autoShortlistScore:
   375→        form.autoShortlistScore.trim() === ""
   376→          ? null
   377→          : Number(form.autoShortlistScore),
   378→      autoShortlistStage:
   379→        cleanedStages.length === 0 ? null : form.autoShortlistStage || null,
   380→      applyTemplate: form.applyTemplate.trim() || null,
   381→      acceptTemplate: form.acceptTemplate.trim() || null,
   382→      rejectTemplate: form.rejectTemplate.trim() || null,
   383→      assignmentTitle: form.assignmentTitle.trim() || null,
   384→      assignmentUrl: form.assignmentUrl.trim() || null,
   385→      assignmentNote: form.assignmentNote.trim() || null,
   386→      rubricCriteria: form.rubricCriteria.map((r) => r.trim()).filter(Boolean),
   387→      checklistTemplate: form.checklistTemplate.map((c) => c.trim()).filter(Boolean),
   388→      noteTemplates: form.noteTemplates.map((n) => n.trim()).filter(Boolean),
   389→    };
   390→    if (form.order.trim() !== "") payload.order = Number(form.order);
   391→
   392→    try {
   393→      if (editing) {
   394→        await apiPatch<Position>(`/api/admin/positions/${editing.id}`, payload);
   395→        toast.success("Posisi diperbarui");
   396→      } else {
   397→        await apiPost<Position>("/api/admin/positions", payload);
   398→        toast.success("Posisi ditambahkan");
   399→      }
   400→      onOpenChange(false);
   401→      onSaved();
   402→    } catch (err) {
   403→      reportError(err);
   404→    } finally {
   405→      setSaving(false);
   406→    }
   407→  }
   408→
   409→  function handleUploadCover(file: File) {
   410→    if (!COVER_MIME.includes(file.type)) {
   411→      toast.error("Format gambar harus PNG, JPEG, atau WebP.");
   412→      return;
   413→    }
   414→    if (file.size > COVER_MAX_BYTES) {
   415→      toast.error("Ukuran gambar maksimal 3 MB.");
   416→      return;
   417→    }
   418→    setUploadingCover(true);
   419→    const fd = new FormData();
   420→    fd.append("file", file);
   421→    apiFetch<AdminUploadResponse>("/api/admin/upload", {
   422→      method: "POST",
   423→      body: fd,
   424→    })
   425→      .then((res) => {
   426→        set("coverFileId", res.fileId);
   427→        toast.success("Cover berhasil diunggah");
   428→      })
   429→      .catch((err) => reportError(err))
   430→      .finally(() => {
   431→        setUploadingCover(false);
   432→        if (fileInputRef.current) fileInputRef.current.value = "";
   433→      });
   434→  }
   435→
   436→  function handleAiCover() {
   437→    if (!editing || aiCoverLoading) return;
   438→    setAiCoverLoading(true);
   439→    const tid = toast.loading(
   440→      "Membuat cover dengan AI... proses bisa memakan waktu hingga 1 menit."
   441→    );
   442→    apiPost<AiCoverResponse>("/api/admin/ai/cover", { positionId: editing.id })
   443→      .then((res) => {
   444→        if (res.ok) {
   445→          set("coverFileId", res.fileId);
   446→          toast.success("Cover AI berhasil dibuat", { id: tid });
   447→          onSaved();
   448→        } else {
   449→          toast.error(res.error, { id: tid });
   450→        }
   451→      })
   452→      .catch((err) => {
   453→        reportError(err);
   454→        toast.dismiss(tid);
   455→      })
   456→      .finally(() => setAiCoverLoading(false));
   457→  }
   458→
   459→  return (
   460→    <Dialog open={open} onOpenChange={onOpenChange}>
   461→      <DialogContent className="max-h-[94vh] overflow-hidden rounded-2xl sm:max-w-3xl">
   462→        <DialogHeader>
   463→          <DialogTitle>{editing ? "Edit Posisi" : "Tambah Posisi"}</DialogTitle>
   464→          <DialogDescription>
   465→            {editing
   466→              ? "Perbarui semua pengaturan posisi lowongan."
   467→              : "Tambahkan posisi lowongan baru untuk halaman publik."}
   468→          </DialogDescription>
   469→        </DialogHeader>
   470→
   471→        <div className="-mr-2 max-h-[68vh] overflow-y-auto pr-2 nice-scrollbar sm:max-h-[70vh]">
   472→          <form
   473→            className="flex flex-col gap-3"
   474→            onSubmit={(e) => {
   475→              e.preventDefault();
   476→              void handleSave();
   477→            }}
   478→          >
   479→            {validationErrors.length > 0 ? (
   480→              <div

   486→                  {validationErrors.map((err) => (
   487→                    <li key={err}>{err}</li>
   488→                  ))}
   489→                </ul>
   490→              </div>

   492→
   493→            <Accordion
   494→              type="multiple"
   495→              defaultValue={["dasar", "publikasi"]}
   496→              className="rounded-xl border px-4"
   497→            >
   498→              {/* a. Dasar */}
   499→              <AccordionItem value="dasar" className="border-b last:border-b-0">
   500→                <AccordionTrigger className="py-3.5 hover:no-underline">
   501→                  <span className="flex items-center gap-2">
   502→                    <Briefcase className="size-4 text-rose-600 dark:text-rose-400" aria-hidden="true" />
   503→                    Dasar
   504→                  </span>
   505→                </AccordionTrigger>
   506→                <AccordionContent className="flex flex-col gap-4 pb-4">
   507→                  <div className="flex flex-col gap-1.5">
