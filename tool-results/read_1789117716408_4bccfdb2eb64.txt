    92→function moveItem<T>(items: T[], index: number, direction: -1 | 1): T[] {
    93→  const next = [...items];
    94→  const target = index + direction;
    95→  if (target < 0 || target >= next.length) return next;
    96→  const [item] = next.splice(index, 1);
    97→  next.splice(target, 0, item);
    98→  return next;
    99→}
