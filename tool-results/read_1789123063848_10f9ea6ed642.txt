    93→// Status "halaman sudah digulir" via useSyncExternalStore (murah, tanpa setState di effect).
    94→function useScrolled(threshold = 8): boolean {
    95→  return useSyncExternalStore(
    96→    (notify) => {
    97→      window.addEventListener("scroll", notify, { passive: true });
    98→      return () => window.removeEventListener("scroll", notify);
    99→    },
   100→    () => window.scrollY > threshold,
   101→    () => false,
   102→  );
   103→}

   871→/** JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions). */
   872→function useJobPostingJsonLd(positions: Position[], siteName: string) {
   873→  useEffect(() => {
   874→    const jobPostings = positions.map((position) => ({
   875→      "@context": "https://schema.org",
   876→      "@type": "JobPosting",
   877→      title: position.title,
   878→      description: position.description,
   879→      hiringOrganization: { "@type": "Organization", name: siteName },
   880→      jobLocationType: "TELECOMMUTE",
   881→      employmentType: employmentTypeOf(position.type),
   882→      datePosted: position.createdAt,
   883→      ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   884→    }));
   885→    const script = document.createElement("script");
   886→    script.id = "jobposting-ld";
   887→    script.type = "application/ld+json";
   888→    script.textContent = JSON.stringify(jobPostings);
   889→    document.head.appendChild(script);
   890→    return () => {
   891→      document.getElementById("jobposting-ld")?.remove();
   892→    };
   893→  }, [positions, siteName]);
   894→}
