    84→/** Badge featured (Pin "Unggulan") + chip status otomatis, maks 3 chip + "+n". */
    85→function PositionBadges({
    86→  position,
    87→  stats,
    88→  className,
    89→}: {
    90→  position: Position;
    91→  stats?: PositionPublicStats;
    92→  className?: string;
    93→}) {
    94→  const { t } = useLang();
    95→
    96→  const remaining = stats?.remainingQuota ?? null;
    97→  const chips: StatusChip[] = [
    98→    ...(position.urgent
    99→      ? [
   100→          {
   101→            key: "urgent",
   102→            label: t.positions.urgent,
   103→            className: CHIP_URGENT,
   104→            Icon: Flame,
   105→          },
   106→        ]
   107→      : []),
   108→    ...(isWithinDaysBack(position.createdAt, 7)
   109→      ? [
   110→          {
   111→            key: "baru",
   112→            label: t.positions.baru,
   113→            className: CHIP_AMBER,
   114→            Icon: Sparkles,
   115→          },
   116→        ]
   117→      : []),
   118→    ...(position.closesAt && isWithinDaysAhead(position.closesAt, 3)
   119→      ? [
   120→          {
   121→            key: "closing-soon",
   122→            label: t.positions.segeraDitutup,
   123→            className: CHIP_ORANGE,
   124→            Icon: Timer,
   125→          },
   126→        ]
   127→      : []),
   128→    ...(remaining === 0
   129→      ? [
   130→          {
   131→            key: "quota-full",
   132→            label: t.positions.kuotaPenuh,
   133→            className: CHIP_ZINC,
   134→            Icon: Users,
   135→          },
   136→        ]
   137→      : []),
   138→    ...(remaining !== null && remaining > 0 && remaining <= 3
   139→      ? [
   140→          {
   141→            key: "quota-low",
   142→            label: fillTemplate(t.positions.sisaKuota, { n: remaining }),
   143→            className: CHIP_AMBER,
   144→            Icon: Users,
   145→          },
   146→        ]
   147→      : []),
   148→  ];
   149→
   150→  if (!position.featured && chips.length === 0) return null;
   151→
   152→  const visibleChips = chips.slice(0, MAX_STATUS_CHIPS);
   153→  const hiddenCount = chips.length - visibleChips.length;
   154→
   155→  return (
   156→    <div className={cn("flex flex-wrap items-center gap-1.5", className)}>
   157→      {position.featured ? (
   158→        <Badge className={CHIP_FEATURED} variant="outline">
   159→          <Pin className="h-3 w-3" aria-hidden="true" />
   160→          {t.positions.featured}
   161→        </Badge>
   162→      ) : null}
   163→      {visibleChips.map((chip) => (
   164→        <Badge key={chip.key} className={chip.className} variant="outline">
   165→          <chip.Icon className="h-3 w-3" aria-hidden="true" />
   166→          {chip.label}
   167→        </Badge>
   168→      ))}
   169→      {hiddenCount > 0 ? (
   170→        <Badge className={CHIP_MUTED} variant="outline">
   171→          {fillTemplate(t.positions.chipMore, { n: hiddenCount })}
   172→        </Badge>
   173→      ) : null}
   174→    </div>
   175→  );
   176→}

   263→              <Badge
   264→                key={`${position.id}-benefit-${index}`}

   266→                className="rounded-full font-normal"
   267→              >
   268→                {benefit}
   269→              </Badge>
   270→            ))}
   271→            {position.benefits.length > MAX_BENEFIT_PREVIEW ? (
   272→              <span className="text-xs text-muted-foreground">
   273→                {fillTemplate(t.positions.benefitLainnya, {
   274→                  n: position.benefits.length - MAX_BENEFIT_PREVIEW,
   275→                })}
   276→              </span>
   277→            ) : null}
   278→          </div>
   279→        ) : null}
   280→
   281→        {position.requirements.length > 0 ? (
   282→          <ul className="space-y-2">
   283→            {position.requirements.slice(0, 3).map((requirement, index) => (
   284→              <li
   285→                key={`${position.id}-req-${index}`}
   286→                className="flex items-start gap-2 text-sm text-muted-foreground"
   287→              >
   288→                <Check
   289→                  className="mt-0.5 h-4 w-4 shrink-0 text-emerald-600 dark:text-emerald-400"
   290→                  aria-hidden="true"
   291→                />
   292→                <span>{requirement}</span>
   293→              </li>
   294→            ))}
   295→          </ul>
   296→        ) : (
   297→          <p className="text-sm italic text-muted-foreground">
   298→            {t.positions.reqFallback}
   299→          </p>
   300→        )}
   301→
   302→        {position.closesAt && !closed ? (
   303→          <DeadlineCountdownCompact deadline={position.closesAt} />
   304→        ) : null}
   305→
   306→        <div className="mt-auto flex items-center justify-between gap-3 border-t pt-4">
   307→          <span className="text-xs text-muted-foreground">
   308→            {position.requirements.length > 0
   309→              ? `${position.requirements.length} ${t.positions.requirementsCount}`
   310→              : ""}
   311→          </span>
   312→          <div className="flex items-center gap-2">
   313→            <Button
   314→              size="sm"
   315→              variant="outline"
   316→              aria-label={`${t.positions.detailAria}: ${position.title}`}
   317→              onClick={(event) => {
   318→                event.stopPropagation();
   319→                onOpenDetail(position);
   320→              }}
   321→            >
   322→              <Eye className="h-4 w-4" aria-hidden="true" />
   323→              {t.positions.detail}
   324→            </Button>
   325→            {canApply ? (
   326→              <Button
   327→                size="sm"
   328→                disabled={applyDisabled}
   329→                className="transition-transform hover:scale-[1.04] active:scale-95"
   330→                onClick={(event) => {
   331→                  event.stopPropagation();
   332→                  onApply(position.id);
   333→                }}
   334→              >
   335→                {applyLabel}
   336→              </Button>
   337→            ) : null}
   338→          </div>
   339→        </div>
   340→      </div>
   341→    </Card>
   342→  );
   343→}

   345→/** Baris share kit posisi: WhatsApp (caption lengkap + UTM), Salin Link, QR Code. */
   346→function PositionShareRow({
   347→  position,
   348→  siteName,
   349→  onOpenQr,
   350→}: {
   351→  position: Position;
   352→  siteName: string;
   353→  onOpenQr: () => void;
   354→}) {
   355→  const { t } = useLang();
   356→
   357→  const shareLink = buildPositionUrl(position, "whatsapp");
   358→  const waCaption = [
   359→    fillTemplate(t.positions.waCaption, { title: position.title, siteName }),
   360→    position.salaryVisible && position.salaryText
   361→      ? position.salaryText
   362→      : null,
   363→    `${t.positions.waDaftar} ${shareLink}`,
   364→  ]
   365→    .filter(Boolean)
   366→    .join(" — ");
   367→
   368→  async function copyLink() {
   369→    try {
   370→      await navigator.clipboard.writeText(buildPositionUrl(position, "copy"));
   371→      toast.success(t.positions.salinBerhasil);
   372→    } catch {
   373→      toast.error(t.positions.salinGagal);
   374→    }
   375→  }
   376→
   377→  return (
   378→    <div className="rounded-xl border bg-muted/40 p-3">
   379→      <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   380→        {t.positions.bagikan}
   381→      </p>
   382→      <div className="mt-2 flex flex-wrap gap-2">
   383→        <Button variant="outline" size="sm" className="h-11 sm:h-9" asChild>
   384→          <a href={waShareHref(waCaption)} target="_blank" rel="noopener noreferrer">
   385→            <MessageCircle className="h-4 w-4" aria-hidden="true" />
   386→            {t.positions.bagikanWa}
   387→          </a>
   388→        </Button>
   389→        <Button
   390→          variant="outline"
   391→          size="sm"
   392→          className="h-11 sm:h-9"
   393→          onClick={() => void copyLink()}
   394→        >
   395→          <Copy className="h-4 w-4" aria-hidden="true" />
   396→          {t.positions.bagikanSalin}
   397→        </Button>
   398→        <Button
   399→          variant="outline"
   400→          size="sm"
   401→          className="h-11 sm:h-9"
   402→          onClick={onOpenQr}
   403→        >
   404→          <QrCode className="h-4 w-4" aria-hidden="true" />
   405→          {t.positions.bagikanQr}
   406→        </Button>
   407→      </div>
   408→    </div>
   409→  );
   410→}

   658→function PositionDetailDialog({
   659→  position,
   660→  siteName,
   661→  stats,
   662→  canApply,
   663→  onClose,
   664→  onApply,
   665→}: {
   666→  position: Position | null;
   667→  siteName: string;
   668→  stats?: PositionPublicStats;
   669→  canApply: boolean;
   670→  onClose: () => void;
   671→  onApply: (positionId: string) => void;
   672→}) {
   673→  const { t } = useLang();
   674→  const [qrOpen, setQrOpen] = useState(false);
   675→  const [qrDataUrl, setQrDataUrl] = useState<string | null>(null);
   676→  const [qrLoading, setQrLoading] = useState(false);
   677→
   678→  // Views: hitung 1x per sesi per posisi (fire-and-forget, gagal diam).
   679→  useEffect(() => {
   680→    if (!position) return;
   681→    let viewed = false;
   682→    try {
   683→      const key = `viewed-${position.id}`;
   684→      viewed = window.sessionStorage.getItem(key) === "1";
   685→      if (!viewed) window.sessionStorage.setItem(key, "1");
   686→    } catch {
   687→      viewed = false; // storage diblokir — tetap kirim
   688→    }
   689→    if (viewed) return;
   690→    void fetch(`/api/positions/${position.id}/view`, { method: "POST" }).catch(
   691→      () => {},
   692→    );
   693→  }, [position]);
   694→
   695→  // QR deep link digenerate saat dialog QR dibuka (pola share-menu).
   696→  useEffect(() => {
   697→    if (!qrOpen || !position) return;
   698→    let cancelled = false;
   699→    QRCode.toDataURL(buildPositionUrl(position), { width: 320, margin: 2 })
   700→      .then((dataUrl) => {
   701→        if (cancelled) return;
   702→        setQrDataUrl(dataUrl);
   703→        setQrLoading(false);
   704→      })
   705→      .catch(() => {
   706→        if (cancelled) return;
   707→        setQrDataUrl(null);
   708→        setQrLoading(false);
   709→      });
   710→    return () => {
   711→      cancelled = true;
   712→    };
   713→  }, [qrOpen, position]);
   714→
   715→  // Loading di-set dari event handler (bukan body effect) — pola share-menu.
   716→  function openQr() {
   717→    setQrLoading(true);
   718→    setQrDataUrl(null);
   719→    setQrOpen(true);
   720→  }
   721→
   722→  return (
   723→    <>
   724→      <Dialog open={position !== null} onOpenChange={(open) => !open && onClose()}>
   725→        <DialogContent
   726→          showCloseButton={false}
   727→          className="max-h-[92vh] max-w-2xl gap-0 overflow-y-auto nice-scrollbar p-0"
   728→        >
   729→          {position ? (
   730→            <>
   731→              <DialogClose
   732→                className={cn(
   733→                  "absolute top-3 right-3 z-10 flex h-9 w-9 items-center justify-center rounded-full bg-background/80 shadow-sm backdrop-blur transition-colors hover:bg-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   734→                  !position.coverFileId && "border",
   735→                )}
   736→                aria-label={t.positions.tutup}
   737→              >
   738→                <X className="h-4 w-4" aria-hidden="true" />
   739→              </DialogClose>
   740→              <PositionDialogInner
   741→                position={position}
   742→                siteName={siteName}
   743→                stats={stats}
   744→                canApply={canApply}
   745→                onApply={onApply}
   746→                onOpenQr={openQr}
   747→              />
   748→            </>
   749→          ) : null}
   750→        </DialogContent>
   751→      </Dialog>
   752→
   753→      {position ? (
   754→        <Dialog open={qrOpen} onOpenChange={setQrOpen}>
   755→          <DialogContent className="max-w-xs sm:max-w-sm">
   756→            <DialogHeader>
   757→              <DialogTitle>{t.positions.qrTitle}</DialogTitle>
   758→              <DialogDescription>{t.positions.qrCaption}</DialogDescription>
   759→            </DialogHeader>
   760→            <div className="flex items-center justify-center rounded-xl border bg-white p-4">
   761→              {qrLoading ? (
   762→                <Loader2
   763→                  className="h-8 w-8 animate-spin text-muted-foreground"
   764→                  aria-hidden="true"
   765→                />
   766→              ) : qrDataUrl ? (
   767→                <img
   768→                  src={qrDataUrl}
   769→                  alt={t.positions.qrAlt}
   770→                  className="h-56 w-56"
   771→                  width={224}
   772→                  height={224}
   773→                />
   774→              ) : (
   775→                <p className="text-sm text-muted-foreground">
   776→                  {t.positions.salinGagal}
   777→                </p>
   778→              )}
   779→            </div>
   780→          </DialogContent>
   781→        </Dialog>
   782→      ) : null}
   783→    </>
   784→  );
   785→}

   812→export function PositionsSection({
   813→  positions,
   814→  siteName,
   815→  canApply,
   816→  onApply,
   817→  positionStats = {},
   818→  initialSlug = null,
   819→}: {
   820→  positions: Position[];
   821→  siteName: string;
   822→  canApply: boolean; // false = sections.applyForm nonaktif, semua tombol lamar disembunyikan
   823→  onApply: (positionId: string) => void;
   824→  /** Kuota & jumlah lamaran per id posisi (dari PublicContentResponse.positionStats). */
   825→  positionStats?: Record<string, PositionPublicStats>;
   826→  /** Slug dari deep link /?posisi=slug — dialog detail posisi dibuka otomatis. */
   827→  initialSlug?: string | null;
   828→}) {
   829→  const { t } = useLang();
   830→  const [department, setDepartment] = useState<string | null>(null);
   831→  const [jobType, setJobType] = useState<string | null>(null);
   832→  // Deep link: buka dialog detail posisi otomatis saat pertama render (posisi sudah tersedia).
   833→  const [detail, setDetail] = useState<Position | null>(() => {
   834→    if (!initialSlug) return null;
   835→    return positions.find((p) => p.slug === initialSlug) ?? null;
   836→  });
   837→
   838→  // Featured lebih dulu, lalu urutan existing (Array.sort stabil).
   839→  const sortedPositions = useMemo(
   840→    () => [...positions].sort((a, b) => Number(b.featured) - Number(a.featured)),
   841→    [positions],
   842→  );
   843→
   844→  const departments = useMemo(
   845→    () => Array.from(new Set(sortedPositions.map((p) => p.department))).sort(),
   846→    [sortedPositions],
   847→  );
   848→  const jobTypes = useMemo(
   849→    () => Array.from(new Set(sortedPositions.map((p) => p.type))),
   850→    [sortedPositions],
   851→  );
   852→
   853→  const filtered = sortedPositions.filter(
   854→    (p) =>
   855→      (department === null || p.department === department) &&
   856→      (jobType === null || p.type === jobType),
   857→  );
   858→
   859→  const hasFilters = department !== null || jobType !== null;
   860→
   861→  return (
   862→    <section id="posisi" className="scroll-mt-24 bg-background py-16 md:py-24">
   863→      <Container>
   864→        <FadeIn className="max-w-2xl">
   865→          <Badge variant="outline" className={ROSE_BADGE}>
   866→            {t.positions.badge}
   867→          </Badge>
   868→          <h2 className="mt-4 text-3xl font-bold tracking-tight md:text-4xl">
   869→            {t.positions.title}
   870→          </h2>
   871→          <p className="mt-3 text-muted-foreground">{t.positions.desc}</p>
   872→        </FadeIn>
   873→
   874→        {positions.length === 0 ? (
   875→          <FadeIn className="mt-10">
   876→            <Card className="items-center gap-3 rounded-2xl p-10 text-center">
   877→              <div className="flex h-12 w-12 items-center justify-center rounded-full bg-rose-100 text-rose-600 dark:bg-rose-500/15 dark:text-rose-400">
   878→                <FolderOpen className="h-6 w-6" aria-hidden="true" />
   879→              </div>
   880→              <p className="font-semibold">{t.positions.emptyTitle}</p>
   881→              <p className="text-sm text-muted-foreground">
   882→                {t.positions.emptyBody}
   883→              </p>
   884→            </Card>
   885→          </FadeIn>
   886→        ) : (
   887→          <>
   888→            <FadeIn className="mt-8 space-y-3">
   889→              <div
   890→                role="group"
   891→                aria-label={t.positions.deptLabel}
   892→                className="flex flex-wrap items-center gap-2"
   893→              >
   894→                <FilterChip
   895→                  active={department === null}
   896→                  onClick={() => setDepartment(null)}
   897→                  ariaLabel={`${t.positions.deptLabel}: ${t.positions.all}`}
   898→                >
   899→                  {t.positions.all}
   900→                </FilterChip>
   901→                {departments.map((dept) => (
   902→                  <FilterChip
   903→                    key={dept}
   904→                    active={department === dept}
   905→                    onClick={() => setDepartment(department === dept ? null : dept)}
   906→                    ariaLabel={`${t.positions.deptLabel}: ${dept}`}
   907→                  >
   908→                    {dept}
   909→                  </FilterChip>
   910→                ))}
   911→              </div>
   912→              <div
   913→                role="group"
   914→                aria-label={t.positions.typeLabel}
   915→                className="flex flex-wrap items-center gap-2"
   916→              >
   917→                {jobTypes.map((type) => (
   918→                  <FilterChip
   919→                    key={type}
   920→                    active={jobType === type}
   921→                    onClick={() => setJobType(jobType === type ? null : type)}
   922→                    ariaLabel={`${t.positions.typeLabel}: ${type}`}
   923→                  >
   924→                    <Briefcase className="h-3.5 w-3.5" aria-hidden="true" />
   925→                    {type}
   926→                  </FilterChip>
   927→                ))}
   928→              </div>
   929→            </FadeIn>
   930→
   931→            {filtered.length === 0 ? (
   932→              <FadeIn className="mt-8">
   933→                <Card className="items-center gap-3 rounded-2xl p-10 text-center">
   934→                  <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted text-muted-foreground">
   935→                    <SearchX className="h-6 w-6" aria-hidden="true" />
   936→                  </div>
   937→                  <p className="font-semibold">{t.positions.filterEmptyTitle}</p>
   938→                  <p className="text-sm text-muted-foreground">
   939→                    {t.positions.filterEmptyBody}
   940→                  </p>
   941→                </Card>
   942→              </FadeIn>
   943→            ) : (
   944→              <motion.div
   945→                layout
   946→                className="mt-8 grid grid-cols-1 gap-6 md:grid-cols-2 lg:grid-cols-3"
   947→              >
   948→                <AnimatePresence mode="popLayout" initial={false}>
   949→                  {filtered.map((position) => (
   950→                    <motion.div
   951→                      key={position.id}
   952→                      layout
   953→                      initial={{ opacity: 0, scale: 0.96 }}
   954→                      animate={{ opacity: 1, scale: 1 }}
   955→                      exit={{ opacity: 0, scale: 0.96 }}
   956→                      transition={{ duration: 0.22, ease: "easeOut" }}
   957→                      className="h-full"
   958→                    >
   959→                      <HoverLift className="h-full">
   960→                        <PositionCard
   961→                          position={position}
   962→                          stats={positionStats[position.id]}
   963→                          canApply={canApply}
   964→                          onApply={onApply}
   965→                          onOpenDetail={setDetail}
   966→                        />
   967→                      </HoverLift>
   968→                    </motion.div>
   969→                  ))}
   970→                </AnimatePresence>
   971→              </motion.div>
   972→            )}
   973→          </>
   974→        )}
   975→      </Container>
   976→
   977→      <PositionDetailDialog
   978→        position={detail}
   979→        siteName={siteName}
   980→        stats={detail ? positionStats[detail.id] : undefined}
   981→        canApply={canApply}
   982→        onClose={() => setDetail(null)}
   983→        onApply={onApply}
   984→      />
   985→    </section>
   986→  );
   987→}
