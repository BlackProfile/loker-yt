    82→// Riwayat aktivitas kandidat (timeline vertikal).
    83→// Dipasang dengan key={applicationId} agar state reset saat kandidat berganti.
    84→function ActivityTimeline({ applicationId }: { applicationId: string }) {
    85→  const [logs, setLogs] = useState<LogEntry[]>([]);
    86→  const [loading, setLoading] = useState(true);
    87→
    88→  useEffect(() => {
    89→    let cancelled = false;
    90→    apiGet<LogEntry[]>(
    91→      `/api/admin/logs?applicationId=${encodeURIComponent(applicationId)}&limit=30`
    92→    )
    93→      .then((data) => {
    94→        if (!cancelled) {
    95→          setLogs(data);
    96→          setLoading(false);
    97→        }
    98→      })
    99→      .catch(() => {
   100→        // Timeline bersifat pelengkap; abaikan kegagalan fetch.
   101→        if (!cancelled) setLoading(false);
   102→      });
   103→    return () => {
   104→      cancelled = true;
   105→    };
   106→  }, [applicationId]);
   107→
   108→  if (loading) {
   109→    return (
   110→      <div className="flex items-center gap-2 py-2 text-sm text-muted-foreground">
   111→        <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   112→        Memuat riwayat...
   113→      </div>
   114→    );
   115→  }
   116→
   117→  if (logs.length === 0) {
   118→    return (
   119→      <p className="py-2 text-sm text-muted-foreground">Belum ada aktivitas.</p>
   120→    );
   121→  }
   122→
   123→  return (
   124→    <div className="relative flex max-h-48 flex-col gap-3 overflow-y-auto pl-4 nice-scrollbar">
   125→      <span
   126→        className="absolute top-1.5 left-[4px] h-[calc(100%-12px)] w-px bg-border"
   127→        aria-hidden="true"
   128→      />
   129→      {logs.map((log) => (
   130→        <div key={log.id} className="relative">
   131→          <span
   132→            className="absolute top-1.5 -left-4 size-2 rounded-full bg-rose-500 ring-4 ring-background"
   133→            aria-hidden="true"
   134→          />
   135→          <div className="flex flex-wrap items-center gap-2">
   136→            <span className="text-xs font-medium whitespace-nowrap text-muted-foreground">
   137→              {formatShortDateTime(log.createdAt)}
   138→            </span>
   139→            <Badge
   140→              variant="outline"
   141→              className={`px-1.5 py-0 text-[10px] ${actorBadgeClass(log.actor)}`}
   142→            >
   143→              {log.actor}
   144→            </Badge>
   145→            <Badge variant="secondary" className="px-1.5 py-0 text-[10px]">
   146→              {actionLabel(log.action)}
   147→            </Badge>
   148→          </div>
   149→          {log.detail ? (
   150→            <p className="mt-0.5 text-xs text-muted-foreground">{log.detail}</p>
   151→          ) : null}
   152→        </div>
   153→      ))}
   154→    </div>
   155→  );
   156→}

   158→export function ApplicationDetailDialog({
   159→  application,
   160→  onOpenChange,
   161→  onSaved,
   162→  onDeleted,
   163→}: {
   164→  application: Application | null;
   165→  onOpenChange: (open: boolean) => void;
   166→  onSaved: (app: Application) => void;
   167→  onDeleted: (id: string) => void;
   168→}) {
   169→  const { canMutate, reportError } = useAdminSession();
   170→  const [editStatus, setEditStatus] = useState<ApplicationStatus>("NEW");
   171→  const [editNotes, setEditNotes] = useState("");
   172→  const [saving, setSaving] = useState(false);
   173→  const [deleting, setDeleting] = useState(false);
   174→  const [confirmOpen, setConfirmOpen] = useState(false);
   175→
   176→  // Editor jadwal, tags, rating.
   177→  const [interviewInput, setInterviewInput] = useState("");
   178→  const [savingInterview, setSavingInterview] = useState(false);
   179→  const [tagInput, setTagInput] = useState("");
   180→  const [tagsSaving, setTagsSaving] = useState(false);
   181→  const [ratingSaving, setRatingSaving] = useState(false);
   182→
   183→  useEffect(() => {
   184→    if (application) {
   185→      setEditStatus(application.status);
   186→      setEditNotes(application.adminNotes ?? "");
   187→      setInterviewInput(isoToLocalInput(application.interviewAt));
   188→      setTagInput("");
   189→      setSaving(false);
   190→      setDeleting(false);
   191→      setConfirmOpen(false);
   192→    }
   193→  }, [application]);
   194→
   195→  if (!application) return null;
   196→
   197→  const app = application;
   198→
   199→  async function patch(
   200→    body: Record<string, unknown>,
   201→    successMessage: string
   202→  ): Promise<Application | null> {
   203→    if (!canMutate) {
   204→      toast.error("Anda tidak memiliki akses untuk aksi ini.");
   205→      return null;
   206→    }
   207→    try {
   208→      const updated = await apiPatch<Application>(
   209→        `/api/admin/applications/${app.id}`,
   210→        body
   211→      );
   212→      toast.success(successMessage);
   213→      onSaved(updated);
   214→      return updated;
   215→    } catch (err) {
   216→      reportError(err);
   217→      return null;
   218→    }
   219→  }
   220→
   221→  async function handleSave() {
   222→    if (saving) return;
   223→    setSaving(true);
   224→    try {
   225→      const updated = await apiPatch<Application>(
   226→        `/api/admin/applications/${app.id}`,
   227→        { status: editStatus, adminNotes: editNotes }
   228→      );
   229→      toast.success("Perubahan disimpan");
   230→      onSaved(updated);
   231→      onOpenChange(false);
   232→    } catch (err) {
   233→      reportError(err);
   234→    } finally {
   235→      setSaving(false);
   236→    }
   237→  }
   238→
   239→  async function handleDelete() {
   240→    if (deleting) return;
   241→    setDeleting(true);
   242→    try {
   243→      await apiDelete<{ ok: boolean }>(`/api/admin/applications/${app.id}`);
   244→      toast.success("Lamaran dihapus");
   245→      onDeleted(app.id);
   246→      onOpenChange(false);
   247→    } catch (err) {
   248→      reportError(err);
   249→    } finally {
   250→      setDeleting(false);
   251→      setConfirmOpen(false);
   252→    }
   253→  }
   254→
   255→  async function handleSaveInterview() {
   256→    if (savingInterview) return;
   257→    if (!interviewInput) {
   258→      toast.error("Pilih tanggal dan jam terlebih dahulu.");
   259→      return;
   260→    }
   261→    const iso = localInputToIso(interviewInput);
   262→    if (!iso) {
   263→      toast.error("Tanggal tidak valid.");
   264→      return;
   265→    }
   266→    setSavingInterview(true);
   267→    await patch({ interviewAt: iso }, "Jadwal wawancara disimpan");
   268→    setSavingInterview(false);
   269→  }
   270→
   271→  async function handleClearInterview() {
   272→    if (savingInterview) return;
   273→    setSavingInterview(true);
   274→    await patch({ interviewAt: null }, "Jadwal wawancara dihapus");
   275→    setSavingInterview(false);
   276→  }
   277→
   278→  async function handleAddTag(e: FormEvent<HTMLFormElement>) {
   279→    e.preventDefault();
   280→    const tag = tagInput.trim();
   281→    if (!tag || tagsSaving) return;
   282→    if (app.tags.includes(tag)) {
   283→      toast.error("Tag sudah ada.");
   284→      return;
   285→    }
   286→    setTagsSaving(true);
   287→    const updated = await patch({ tags: [...app.tags, tag] }, "Tag ditambahkan");
   288→    if (updated) setTagInput("");
   289→    setTagsSaving(false);
   290→  }
   291→
   292→  async function handleRemoveTag(tag: string) {
   293→    if (tagsSaving) return;
   294→    setTagsSaving(true);
   295→    await patch(
   296→      { tags: app.tags.filter((t) => t !== tag) },
   297→      "Tag dihapus"
   298→    );
   299→    setTagsSaving(false);
   300→  }
   301→
   302→  async function handleRating(rating: number) {
   303→    if (ratingSaving) return;
   304→    setRatingSaving(true);
   305→    await patch({ rating }, `Rating disimpan (${rating}/5)`);
   306→    setRatingSaving(false);
   307→  }
   308→
   309→  async function handleTalentPool(talentPool: boolean) {
   310→    await patch(
   311→      { talentPool },
   312→      talentPool ? "Ditambahkan ke Talent Pool" : "Dikeluarkan dari Talent Pool"
   313→    );
   314→  }
   315→
   316→  async function handleCopyTracking() {
   317→    const ok = await copyText(app.trackingCode);
   318→    if (ok) toast.success("Kode pelacakan disalin");
   319→    else toast.error("Gagal menyalin ke clipboard");
   320→  }
   321→
   322→  return (
   323→    <Dialog open onOpenChange={onOpenChange}>
   324→      <DialogContent className="max-h-[92vh] overflow-hidden rounded-2xl sm:max-w-3xl">
   325→        <DialogHeader>
   326→          <DialogTitle className="flex flex-wrap items-center gap-2 pr-6 text-lg font-bold">
   327→            <span>{app.name}</span>
   328→            <StatusBadge status={app.status} />
   329→            {app.talentPool ? (
   330→              <Badge
   331→                variant="outline"
   332→                className="border-emerald-200 bg-emerald-100 text-emerald-700 dark:border-emerald-900 dark:bg-emerald-950 dark:text-emerald-400"
   333→              >
   334→                Talent Pool
   335→              </Badge>
   336→            ) : null}
   337→          </DialogTitle>
   338→          <DialogDescription className="flex flex-wrap items-center gap-1.5">
   339→            <span className="sr-only">Detail lamaran, catatan admin, ubah status, dan riwayat aktivitas.</span>
   340→            <span className="text-muted-foreground">Kode pelacakan</span>
   341→            <code className="rounded bg-muted px-1.5 py-0.5 font-mono text-xs text-foreground">
   342→              {app.trackingCode}
   343→            </code>
   344→            <button
   345→              type="button"
   346→              onClick={() => void handleCopyTracking()}
   347→              aria-label="Salin kode pelacakan"
   348→              className="text-muted-foreground outline-none hover:text-foreground focus-visible:ring-2 focus-visible:ring-ring/50"
   349→            >
   350→              <Copy className="size-3.5" aria-hidden="true" />
   351→            </button>
   352→          </DialogDescription>
   353→        </DialogHeader>
   354→
   355→        <div className="-mr-2 max-h-[75vh] overflow-y-auto pr-2 nice-scrollbar">
   356→          <div className="flex flex-col gap-4">
   357→            {/* Panel AI */}
   358→            <AiPanel app={app} onUpdated={onSaved} />
   359→
   360→            {/* Info grid */}
   361→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   362→              <InfoItem label="Email">
   363→                <a
   364→                  href={`mailto:${app.email}`}
   365→                  className="hover:text-rose-600 underline-offset-2 hover:underline"
   366→                >
   367→                  {app.email}
   368→                </a>
   369→              </InfoItem>
   370→              <InfoItem label="WhatsApp">
   371→                <a
   372→                  href={waHref(app.phone)}
   373→                  target="_blank"
   374→                  rel="noopener noreferrer"
   375→                  className="hover:text-rose-600 underline-offset-2 hover:underline"
   376→                >
   377→                  {app.phone || "-"}
   378→                </a>
   379→              </InfoItem>
   380→              <InfoItem label="Posisi">{app.positionTitle ?? "-"}</InfoItem>
   381→              <InfoItem label="Tanggal Daftar">
   382→                {formatDateTime(app.createdAt)}
   383→              </InfoItem>
   384→              <InfoItem label="Portofolio">
   385→                {app.portfolioUrl ? (
   386→                  <a
   387→                    href={normalizeUrl(app.portfolioUrl)}
   388→                    target="_blank"
   389→                    rel="noopener noreferrer"
   390→                    className="inline-flex items-center gap-1.5 text-rose-600 underline-offset-2 hover:underline"
   391→                  >
   392→                    <Link2 className="size-3.5 shrink-0" aria-hidden="true" />
   393→                    <span className="truncate">{app.portfolioUrl}</span>
   394→                  </a>
   395→                ) : (
   396→                  <span className="text-muted-foreground">-</span>
   397→                )}
   398→              </InfoItem>
   399→              <InfoItem label="Sosial Media">
   400→                {app.socialLinks ? (
   401→                  <a
   402→                    href={normalizeUrl(app.socialLinks)}
   403→                    target="_blank"
   404→                    rel="noopener noreferrer"
   405→                    className="inline-flex items-center gap-1.5 text-rose-600 underline-offset-2 hover:underline"
   406→                  >
   407→                    <Link2 className="size-3.5 shrink-0" aria-hidden="true" />
   408→                    <span className="truncate">{app.socialLinks}</span>
   409→                  </a>
   410→                ) : (
   411→                  <span className="text-muted-foreground">-</span>
   412→                )}
   413→              </InfoItem>
   414→            </div>
   415→
   416→            {/* Jadwal wawancara */}
   417→            <div className="rounded-lg border p-3">
   418→              <div className="mb-2 flex items-center gap-2">
   419→                <CalendarClock className="size-4 text-orange-500" aria-hidden="true" />
   420→                <p className="text-sm font-semibold">Jadwal Wawancara</p>
   421→              </div>
   422→              {app.interviewAt ? (
   423→                <p className="mb-2 text-sm text-muted-foreground">
   424→                  Terjadwal:{" "}
   425→                  <span className="font-medium text-foreground">
   426→                    {formatDateTime(app.interviewAt)}
   427→                  </span>
   428→                </p>
   429→              ) : (
   430→                <p className="mb-2 text-sm text-muted-foreground">
   431→                  Belum ada jadwal.
   432→                </p>

   434→              {canMutate ? (
   435→                <div className="flex flex-wrap items-center gap-2">
   436→                  <Input
   437→                    type="datetime-local"
   438→                    value={interviewInput}
   439→                    onChange={(e) => setInterviewInput(e.target.value)}
   440→                    aria-label="Atur tanggal dan jam wawancara"
   441→                    className="h-9 w-fit"

   443→                  <Button

   446→                    onClick={() => void handleSaveInterview()}

   479→                    </span>

   481→                      href={`/api/files/${app.cvFileId}`}
   482→                      target="_blank"
   483→                      rel="noopener noreferrer"
   484→                      className="inline-flex h-8 items-center rounded-md border px-2.5 text-xs font-medium outline-none hover:bg-accent focus-visible:ring-2 focus-visible:ring-ring/50"
   485→                    >
   486→                      Unduh
   487→                    </a>
   488→                  </div>
   489→                ) : null}
   490→                {app.introFileId ? (
   491→                  <div className="flex flex-col gap-2">
   492→                    <div className="flex items-center gap-2">
   493→                      <AudioLines className="size-4 shrink-0 text-muted-foreground" aria-hidden="true" />
   494→                      <span className="min-w-0 flex-1 truncate text-sm">

   504→                    {app.transcript ? (
   505→                      <Collapsible>
   506→                        <CollapsibleTrigger className="group flex w-fit items-center gap-1 text-xs font-medium text-muted-foreground outline-none hover:text-foreground focus-visible:ring-2 focus-visible:ring-ring/50">

   510→                          />
   511→                          Lihat Transkripsi AI

   515→                            {app.transcript}
   516→                          </p>
   517→                        </CollapsibleContent>
   518→                      </Collapsible>
   519→                    ) : (
   520→                      <p className="text-xs text-muted-foreground">
   521→                        Transkripsi sedang diproses...
   522→                      </p>
   523→                    )}
   524→                  </div>
   525→                ) : null}
   526→              </div>
   527→            ) : null}
   528→
   529→            <Separator />
   530→
   531→            {/* Pengalaman & Alasan */}
   532→            <div className="flex flex-col gap-4">
   533→              <div>
   534→                <h4 className="mb-1 text-sm font-semibold">Pengalaman</h4>
   535→                <p className="text-sm whitespace-pre-line text-muted-foreground">
   536→                  {app.experience || "-"}
   537→                </p>
   538→              </div>
   539→              <div>
   540→                <h4 className="mb-1 text-sm font-semibold">Alasan Bergabung</h4>
   541→                <p className="text-sm whitespace-pre-line text-muted-foreground">

   547→            <Separator />
   548→
   549→            {/* Editor */}

   552→                <Label className="text-sm">Rating</Label>
   553→                <RatingStars
   554→                  value={app.rating}
   555→                  onChange={canMutate ? (n) => void handleRating(n) : undefined}

   575→                            aria-label={`Hapus tag ${tag}`}

   580→                          </button>
   581→                        ) : null}
   582→                      </span>
