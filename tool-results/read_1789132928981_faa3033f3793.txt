    79→function ChartTooltip({
    80→  active,
    81→  payload,
    82→  label,
    83→}: {
    84→  active?: boolean;
    85→  payload?: { value?: number | string }[];
    86→  label?: string | number;
    87→}) {
    88→  if (!active || !payload?.length) return null;
    89→  const raw = payload[0]?.value ?? 0;
    90→  const dateLabel =
    91→    typeof label === "string"
    92→      ? format(parseISO(label), "d MMMM yyyy", { locale: localeId })
    93→      : String(label);
    94→  return (
    95→    <div className="rounded-lg border bg-background px-3 py-2 text-xs shadow-sm">
    96→      <p className="font-medium">{dateLabel}</p>
    97→      <p className="text-muted-foreground">{raw} lamaran</p>
    98→    </div>
    99→  );
   100→}
