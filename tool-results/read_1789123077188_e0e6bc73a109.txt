    51→/** Parse requirements / daftar string dari JSON string menjadi string[]. */
    52→export function parseRequirements(raw: string): string[] {
    53→  try {
    54→    const parsed: unknown = JSON.parse(raw);
    55→    if (!Array.isArray(parsed)) return [];
    56→    return parsed
    57→      .filter((item): item is string => typeof item === "string")
    58→      .map((item) => item.trim())
    59→      .filter((item) => item.length > 0);
    60→  } catch {
    61→    return [];
    62→  }
    63→}

    65→/** Parse daftar pertanyaan screening dari JSON string (aman terhadap nilai rusak). */
    66→export function parseScreeningQuestions(raw: string): ScreeningQuestion[] {
    67→  try {
    68→    const parsed: unknown = JSON.parse(raw);
    69→    if (!Array.isArray(parsed)) return [];
    70→    const result: ScreeningQuestion[] = [];
    71→    for (let i = 0; i < parsed.length; i++) {
    72→      const obj = (parsed[i] && typeof parsed[i] === "object" ? parsed[i] : {}) as Record<string, unknown>;
    73→      const label = typeof obj.label === "string" ? obj.label.trim() : "";
    74→      if (!label) continue;
    75→      const id = typeof obj.id === "string" && obj.id.trim() ? obj.id.trim() : `q${i + 1}`;
    76→      result.push({ id, label, required: obj.required === true });
    77→    }
    78→    return result.slice(0, 10);
    79→  } catch {
    80→    return [];
    81→  }
    82→}

    84→/** Parse objek string sederhana dari JSON (jawaban screening). */
    85→export function parseStringRecord(raw: string | null | undefined): Record<string, string> | null {
    86→  if (!raw) return null;
    87→  try {
    88→    const parsed: unknown = JSON.parse(raw);
    89→    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return null;
    90→    const result: Record<string, string> = {};
    91→    for (const [key, value] of Object.entries(parsed as Record<string, unknown>)) {
    92→      if (typeof key === "string" && key && typeof value === "string") result[key] = value;
    93→      else if (typeof key === "string" && key && typeof value === "number") result[key] = String(value);
    94→    }
    95→    return Object.keys(result).length > 0 ? result : null;
    96→  } catch {
    97→    return null;
    98→  }
    99→}

   101→/** Parse objek angka sederhana dari JSON (skor rubrik 1-5). */
   102→export function parseScoreRecord(raw: string | null | undefined): Record<string, number> | null {
   103→  if (!raw) return null;
   104→  try {
   105→    const parsed: unknown = JSON.parse(raw);
   106→    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return null;
   107→    const result: Record<string, number> = {};
   108→    for (const [key, value] of Object.entries(parsed as Record<string, unknown>)) {
   109→      const num = typeof value === "number" ? value : Number(value);
   110→      if (key && Number.isFinite(num)) result[key] = Math.min(5, Math.max(1, Math.round(num)));
   111→    }
   112→    return Object.keys(result).length > 0 ? result : null;
   113→  } catch {
   114→    return null;
   115→  }
   116→}

   127→/** Ubah judul menjadi slug URL-aman: huruf kecil, tanda hubung, tanpa karakter aneh. */
   128→export function slugifyTitle(title: string): string {
   129→  const base = title
   130→    .toLowerCase()
   131→    .normalize("NFKD")
   132→    .replace(/[\u0300-\u036f]/g, "")
   133→    .replace(/[^a-z0-9]+/g, "-")
   134→    .replace(/^-+|-+$/g, "")
   135→    .slice(0, 60)
   136→    .replace(/-+$/g, "");
   137→  return base || "posisi";
   138→}

   140→/** Cari slug unik (tambah -2, -3, ... bila sudah dipakai posisi lain). */
   141→export async function ensureUniqueSlug(title: string, excludeId?: string): Promise<string> {
   142→  const base = slugifyTitle(title);
   143→  let candidate = base;
   144→  let counter = 2;
   145→  for (;;) {
   146→    const existing = await db.position.findUnique({ where: { slug: candidate }, select: { id: true } });
   147→    if (!existing || existing.id === excludeId) return candidate;
   148→    candidate = `${base}-${counter}`;
   149→    counter += 1;
   150→  }
   151→}

   323→export function sanitizeSections(value: unknown, fallback: SectionVisibility): SectionVisibility {
   324→  const result = { ...fallback };
   325→  if (!value || typeof value !== "object" || Array.isArray(value)) return result;
   326→  const obj = value as Record<string, unknown>;
   327→  for (const key of SECTION_KEYS) {

   329→    if (typeof v === "boolean") result[key] = v;
   330→  }
   331→  return result;

   366→/** Parse JSON string dari Setting "site" menjadi SiteContent (fallback ke DEFAULT_SITE). */
   367→export function parseSiteContent(raw: string | null | undefined): SiteContent {
   368→  if (!raw) {
   369→    return JSON.parse(JSON.stringify(DEFAULT_SITE)) as SiteContent;
   370→  }
   371→  try {
   372→    const parsed: unknown = JSON.parse(raw);
   373→    return sanitizeSiteContent(parsed, DEFAULT_SITE);
   374→  } catch {
   375→    return JSON.parse(JSON.stringify(DEFAULT_SITE)) as SiteContent;
   376→  }

   379→/* ------------------------- Filter & urutan daftar lamaran ------------------------- */
   380→
   381→export type ParsedApplicationFilters = {
   382→  where: Prisma.ApplicationWhereInput;
   383→  orderBy: Prisma.ApplicationOrderByWithRelationInput[];
   384→  valid: boolean;
   385→};
   386→
   387→/**
   388→ * Bangun where/orderBy untuk GET /api/admin/applications & /export dari query params:
   389→ * status, positionId, q, ratingMin, tag, talentPool ("1"/"true"), hasInterview ("1"), sort.
   390→ */
   391→export function parseApplicationFilters(searchParams: URLSearchParams): ParsedApplicationFilters {
   392→  const where: Prisma.ApplicationWhereInput = {};
   393→  let valid = true;
   394→
   395→  const status = searchParams.get("status");
   396→  if (status) {
   397→    // Tahap bisa berupa 5 status bawaan ATAU label tahap kustom milik posisi.
   398→    const clean = status.trim().slice(0, 40);
   399→    if (clean) where.status = clean;
   400→    else valid = false;
   401→  }
   402→
   403→  const source = searchParams.get("source")?.trim();
   404→  if (source) where.source = { contains: source.slice(0, 40) };
   405→
   406→  const positionId = searchParams.get("positionId");
   407→  if (positionId) where.positionId = positionId;
   408→
   409→  const q = searchParams.get("q")?.trim();
   410→  if (q) {
   411→    where.OR = [{ name: { contains: q } }, { email: { contains: q } }];
   412→  }
   413→
   414→  const ratingMin = Number.parseInt(searchParams.get("ratingMin") ?? "", 10);
   415→  if (Number.isInteger(ratingMin)) where.rating = { gte: ratingMin };
   416→
   417→  const tag = searchParams.get("tag")?.trim();
   418→  if (tag) {
   419→    where.tags = { contains: `"${tag.replace(/"/g, "")}"` };
   420→  }
   421→
   422→  const talentPool = searchParams.get("talentPool");
   423→  if (talentPool === "1" || talentPool === "true") where.talentPool = true;
   424→
   425→  const hasInterview = searchParams.get("hasInterview");
   426→  if (hasInterview === "1" || hasInterview === "true") {
   427→    where.NOT = { interviewAt: null };
   428→  }
   429→
   430→  const sort = searchParams.get("sort") ?? "newest";
   431→  let orderBy: Prisma.ApplicationOrderByWithRelationInput[] = [{ createdAt: "desc" }, { id: "desc" }];
   432→  if (sort === "oldest") {
   433→    orderBy = [{ createdAt: "asc" }, { id: "asc" }];
   434→  } else if (sort === "aiScore") {
   435→    orderBy = [{ aiScore: "desc" }, { createdAt: "desc" }];
   436→  }
   437→
   438→  return { where, orderBy, valid };
   439→}

   578→  // 1. Akun admin multi-user (OWNER/HR/VIEWER) jika tabel masih kosong
   579→  const userCount = await db.adminUser.count();

   584→          name: user.name,
   585→          email: user.email,
   586→          passwordHash: hashPassword(user.password),
   587→          role: user.role,
   588→          isActive: true,
   589→        },
   590→      });
   591→    }
   592→  }
   593→
   594→  // 2. Konten situs: buat default atau MERGE dengan data lama
   595→  //    (field baru yang hilang diisi dari DEFAULT_SITE, nilai lama tidak ditimpa)
   596→  const siteSetting = await db.setting.findUnique({ where: { key: "site" } });
   597→  if (!siteSetting) {
   598→    await db.setting.create({ data: { key: "site", value: JSON.stringify(DEFAULT_SITE) } });
   599→  } else {
   600→    try {
   601→      const parsed: unknown = JSON.parse(siteSetting.value);
   602→      const merged = sanitizeSiteContent(parsed, DEFAULT_SITE);
   603→      if (JSON.stringify(merged) !== siteSetting.value) {
   604→        await db.setting.update({ where: { key: "site" }, data: { value: JSON.stringify(merged) } });
   605→      }
   606→    } catch {
   607→      await db.setting.update({ where: { key: "site" }, data: { value: JSON.stringify(DEFAULT_SITE) } });
   608→    }
   609→  }

   714→/**
   715→ * Pastikan data awal (AdminUser, Setting site, posisi, lamaran contoh, tracking code)
   716→ * sudah ada. Idempoten — aman dipanggil di setiap request; pakai cache promise module-level
   717→ * agar tidak race. Gagal me-reset cache agar bisa mencoba ulang.
   718→ */
   719→export async function ensureSeeded(): Promise<void> {
   720→  if (!seedPromise) {
   721→    seedPromise = runSeed().catch((error) => {
   722→      seedPromise = null;
   723→      throw error;
   724→    });
   725→  }
   726→  return seedPromise;
   727→}

   729→/**
   730→ * Auto-tutup posisi (dipanggil lazy dari route GET):
   731→ * 1. closesAt sudah lewat, atau
   732→ * 2. kuota maxApplicants tercapai (lamaran non-ditolak >= kuota).
   733→ * Setiap penutupan dicatat di ActivityLog (actor "Sistem", action AUTO_CLOSE).
   734→ */
   735→export async function closeExpiredPositions(): Promise<void> {
   736→  const now = new Date();
   737→
   738→  const expired = await db.position.findMany({
   739→    where: { isActive: true, closesAt: { lt: now } },
   740→    select: { id: true, title: true },
   741→  });
   742→  for (const position of expired) {
   743→    await db.position.update({ where: { id: position.id }, data: { isActive: false } });
   744→    await db.activityLog.create({
   745→      data: {
   746→        applicationId: null,
   747→        actor: "Sistem",
   748→        action: "AUTO_CLOSE",
   749→        detail: `Posisi "${position.title}" ditutup otomatis karena melewati batas waktu pendaftaran.`,
   750→      },
   751→    });
   752→  }
   753→
   754→  const withQuota = await db.position.findMany({
   755→    where: { isActive: true, maxApplicants: { not: null } },
   756→    select: { id: true, title: true, maxApplicants: true, _count: { select: { applications: { where: { status: { not: "REJECTED" } } } } } },
   757→  });
   758→  for (const position of withQuota) {
   759→    const quota = position.maxApplicants ?? 0;
   760→    if (quota > 0 && position._count.applications >= quota) {
   761→      await db.position.update({ where: { id: position.id }, data: { isActive: false } });
   762→      await db.activityLog.create({
   763→        data: {
   764→          applicationId: null,
   765→          actor: "Sistem",
   766→          action: "AUTO_CLOSE",
   767→          detail: `Posisi "${position.title}" ditutup otomatis karena kuota pelamar (${quota}) sudah penuh.`,
   768→        },
   769→      });
   770→    }
   771→  }
   772→}
