   814→   814→/** JSON-LD JobPosting untuk SEO (dipasang/dibongkar sesuai perubahan positions). */
   815→   815→function useJobPostingJsonLd(positions: Position[], siteName: string) {
   816→   816→  useEffect(() => {
   817→   817→    const jobPostings = positions.map((position) => ({
   818→   818→      "@context": "https://schema.org",
   819→   819→      "@type": "JobPosting",
   820→   820→      title: position.title,
   821→   821→      description: position.description,
   822→   822→      hiringOrganization: { "@type": "Organization", name: siteName },
   823→   823→      jobLocationType: "TELECOMMUTE",
   824→   824→      employmentType: employmentTypeOf(position.type),
   825→   825→      datePosted: position.createdAt,
   826→   826→      ...(position.closesAt ? { validThrough: position.closesAt } : {}),
   827→   827→    }));
   828→   828→    const script = document.createElement("script");
   829→   829→    script.id = "jobposting-ld";
   830→   830→    script.type = "application/ld+json";
   831→   831→    script.textContent = JSON.stringify(jobPostings);
   832→   832→    document.head.appendChild(script);
   833→   833→    return () => {
   834→   834→      document.getElementById("jobposting-ld")?.remove();
   835→   835→    };
   836→   836→  }, [positions, siteName]);
   837→   837→}
